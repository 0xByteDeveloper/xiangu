static const char* GetHexViewName(HexViewType t) {
	switch (t) {
	case HexViewType::ByteHex: return "Byte Hex";
	case HexViewType::ByteDec: return "Byte Dec";
	case HexViewType::Word2Hex: return "2B Hex";
	case HexViewType::Word2Dec: return "2B Dec";
	case HexViewType::Word4Hex: return "4B Hex";
	case HexViewType::Word4Dec: return "4B Dec";
	case HexViewType::Word8Hex: return "8B Hex";
	case HexViewType::Word8Dec: return "8B Dec";
	case HexViewType::Float: return "Float";
	case HexViewType::Double: return "Double";
	default: return "Byte Hex";
	}

static int GetHexViewSize(HexViewType t) {
	switch (t) {
	case HexViewType::ByteHex: case HexViewType::ByteDec: return 1;
	case HexViewType::Word2Hex: case HexViewType::Word2Dec: return 2;
	case HexViewType::Word4Hex: case HexViewType::Word4Dec: case HexViewType::Float: return 4;
	case HexViewType::Word8Hex: case HexViewType::Word8Dec: case HexViewType::Double: return 8;
	default: return 1;
	}
}

static void LoadConfig() {
	std::ifstream f(CONFIG_FILE);
	if (!f) return;
	std::map<std::string, std::string> cfg;
	std::string ln;
	while (std::getline(f, ln)) {
		if (ln.empty() || ln[0] == '[' || ln[0] == '#') continue;
		size_t eq = ln.find('=');
		if (eq != std::string::npos) {
			std::string k = ln.substr(0, eq), v = ln.substr(eq + 1);
			while (!k.empty() && k.back() == ' ') k.pop_back();
			while (!v.empty() && v[0] == ' ') v.erase(0, 1);
			cfg[k] = v;
		}
	}
	f.close();
	LoadThemeConfig(cfg);
	g_fonts.LoadConfig(cfg);
	try {
		if (cfg.count("showModuleNames")) g_showModuleNames = (cfg["showModuleNames"] == "1");
		if (cfg.count("showHexPrefix")) g_showHexPrefix = (cfg["showHexPrefix"] == "1");
		if (cfg.count("hexBytesPerRow")) g_hexBytesPerRow = std::stoi(cfg["hexBytesPerRow"]);
		if (cfg.count("hexScrollMode")) g_hexScrollMode = std::stoi(cfg["hexScrollMode"]);
		if (cfg.count("hexViewType")) g_hexViewType = (HexViewType)std::stoi(cfg["hexViewType"]);
		if (cfg.count("hexAutoRowBytes")) g_hexAutoRowBytes = (cfg["hexAutoRowBytes"] == "1");
		if (cfg.count("useCustomBase")) g_useCustomBase = (cfg["useCustomBase"] == "1");
		if (cfg.count("displayBase")) g_displayBase = std::stoull(cfg["displayBase"]);
		if (cfg.count("disVisibleLines")) g_disVisibleLines = std::stoi(cfg["disVisibleLines"]);
		if (cfg.count("count")) {
			int count = std::stoi(cfg["count"]);
			if (count == (int)g_columnOrder.size()) {
				for (int i = 0; i < count; i++) {
					std::string p = "col" + std::to_string(i);
					if (cfg.count(p + "_type")) g_columnOrder[i].type = (DisasmColumn)std::stoi(cfg[p + "_type"]);
					if (cfg.count(p + "_width")) g_columnOrder[i].width = std::stof(cfg[p + "_width"]);
					if (cfg.count(p + "_visible")) g_columnOrder[i].visible = (cfg[p + "_visible"] == "1");
					const char* names[] = { "Address","Bytes","Jump","Opcode","Comments" };
					g_columnOrder[i].name = names[(int)g_columnOrder[i].type];
				}
			}
		}
	}
	catch (...) {}
	g_hexBytesPerRow = std::clamp(g_hexBytesPerRow, 4, 64);
	g_hexScrollMode = std::clamp(g_hexScrollMode, 0, 1);
	g_disVisibleLines = std::clamp(g_disVisibleLines, 10, 200);
}

// === TOKEN PARSING ===
static bool IsRegister(const std::string& s) {
	std::string l = s; for (auto& c : l) c = tolower(c);
	return g_regs32.count(l) || g_regs64.count(l) || g_regsSeg.count(l) || g_regsXmm.count(l) || g_regs8.count(l) || g_regs16.count(l);
}

static bool IsHexNumber(const std::string& s) {
	if (s.empty()) return false;
	size_t st = 0; if (s[0] == '-') st = 1;
	if (s.size() > st + 2 && s[st] == '0' && (s[st + 1] == 'x' || s[st + 1] == 'X')) { for (size_t i = st + 2; i < s.size(); i++) if (!isxdigit(s[i])) return false; return s.size() > st + 2; }
	if (s.back() == 'h' || s.back() == 'H') { for (size_t i = st; i < s.size() - 1; i++) if (!isxdigit(s[i])) return false; return true; }
	for (size_t i = st; i < s.size(); i++) if (!isxdigit(s[i])) return false;
	return !s.empty() && st < s.size();
}

static bool IsModuleOffset(const std::string& s) {
	size_t dp = s.find('.'); if (dp == std::string::npos) return false;
	std::string ad = s.substr(dp); for (auto& c : ad) c = tolower(c);
	if (ad.find(".dll") != 0 && ad.find(".exe") != 0 && ad.find(".sys") != 0) return false;
	size_t pp = s.find('+'); if (pp == std::string::npos) return true;
	std::string op = s.substr(pp + 1); if (op.empty()) return false;
	if (op.size() >= 2 && op[0] == '0' && (op[1] == 'x' || op[1] == 'X')) return true;
	for (char c : op) if (!isxdigit(c)) return false;
	return !op.empty();
}

struct OperandToken { std::string text; ImU32 color; };

static std::vector<OperandToken> ParseOperandTokens(const std::string& op, bool modNames) {
	std::vector<OperandToken> toks;
	if (op.empty()) return toks;
	std::string cur;
	auto endsWithExt = [](const std::string& s) { if (s.size() < 4) return false; std::string l = s; for (auto& c : l) c = tolower(c); std::string sf = l.substr(l.size() - 4); return sf == ".exe" || sf == ".dll" || sf == ".sys"; };
	auto flush = [&]() {
		if (cur.empty()) return;
		OperandToken t; t.text = cur;
		std::string l = cur; for (auto& c : l) c = tolower(c);
		if (modNames && IsModuleOffset(cur)) t.color = g_colors.Green;
		else if (IsRegister(cur)) t.color = g_colors.Red;
		else if (IsHexNumber(cur)) t.color = g_colors.Blue;
		else if (l == "ptr" || l == "dword" || l == "qword" || l == "word" || l == "byte" || l == "fword" || l == "tbyte" || l == "oword" || l == "xmmword") t.color = g_colors.White;
		else t.color = g_colors.White;
		toks.push_back(t); cur.clear();
	};
	for (size_t i = 0; i < op.size(); i++) {
		char c = op[i];
		if (c == '[' || c == ']' || c == ',' || c == ':' || c == '*') { flush(); toks.push_back({ std::string(1,c), g_colors.White }); }
		else if (c == ' ' || c == '\t') { flush(); toks.push_back({ std::string(1,c), g_colors.White }); }
		else if (c == '+' || c == '-') { if (modNames && !cur.empty() && endsWithExt(cur)) cur += c; else { flush(); toks.push_back({ std::string(1,c), g_colors.White }); } }
		else cur += c;
	}
	flush();
	return toks;
}

// === SYMBOL RESOLUTION ===
struct SymbolResult { std::string name; u64 baseAddr, offset; bool found, isExact; };

static SymbolResult FindSymbolAtAddress(u64 addr) {
	SymbolResult r = { "",0,0,false,false }; auto& c = Core::I(); if (!c.loaded) return r;
	std::string es = c.syms.Resolve(addr);
	if (!es.empty()) { r.name = es; r.baseAddr = addr; r.found = r.isExact = true; return r; }
	u64 ba = 0; std::string bn; int ps = c.pe.is64 ? 8 : 4;
	for (auto& p : c.syms.addr2name) { if (p.first <= addr && addr - p.first < (u64)ps && p.first > ba) { ba = p.first; bn = p.second; } }
	if (!bn.empty()) { r.name = bn; r.baseAddr = ba; r.offset = addr - ba; r.found = true; r.isExact = (r.offset == 0); }
	return r;
}

static std::string FormatAddressWithModule(u64 addr, bool showMod) {
	auto& c = Core::I(); char buf[256];
	if (!showMod || !c.loaded) return FormatDisplayAddress(addr, c.pe.is64);
	SymbolResult sym = FindSymbolAtAddress(addr);
	if (sym.found) { if (sym.isExact) return sym.name; sprintf(buf, "%s+%s", sym.name.c_str(), FormatModuleOffset(sym.offset).c_str()); return buf; }
	if (addr >= c.Base() && addr < c.Base() + c.Size()) { sprintf(buf, "%s+%s", g_loadedFileName.c_str(), FormatModuleOffset(addr - c.Base()).c_str()); return buf; }
	sprintf(buf, g_showHexPrefix ? "0x%llX" : "%llX", addr); return buf;
}

static std::string FormatOperandWithModule(const std::string& orig, u64 tgt, u64 memTgt) {
	auto& c = Core::I(); if (!g_showModuleNames || !c.loaded) return orig;
	u64 addr = tgt ? tgt : memTgt; if (addr < 0x1000) return orig;
	std::regex hr("0x([0-9a-fA-F]{4,})|([0-9a-fA-F]{4,})h"); std::smatch m;
	if (std::regex_search(orig, m, hr)) {
		std::string fh = m[0].str(); u64 fv = 0;
		if (fh.substr(0, 2) == "0x" || fh.substr(0, 2) == "0X") fv = strtoull(fh.c_str() + 2, nullptr, 16);
		else fv = strtoull(fh.substr(0, fh.length() - 1).c_str(), nullptr, 16);
		if (fv == addr) return orig.substr(0, m.position()) + FormatAddressWithModule(addr, true) + orig.substr(m.position() + m.length());
	}
	return orig;
}

// === DISASM PRE-BUFFER ===
static void BuildDisasmPreBuffer(u64 tgt, bool force = false) {
	auto& c = Core::I(); if (!c.loaded) return;
	if (!force && g_disPreBufferValid && !g_disPreBuffer.empty()) {
		int ti = -1; for (size_t i = 0; i < g_disPreBuffer.size(); i++) if (g_disPreBuffer[i].addr == tgt) { ti = (int)i; break; }
		if (ti >= 30) return;
	}
	u64 ss = (tgt > 300) ? tgt - 300 : 0;
	std::vector<DisLine> br; u64 bsa = ss; int bti = -1;
	for (u64 ts = ss; ts < tgt && ts < ss + 200; ts++) {
		auto ls = c.Dis(ts, g_disPreBufferLines + 100);
		int ti = -1; for (size_t i = 0; i < ls.size(); i++) { if (ls[i].addr == tgt) { ti = (int)i; break; } if (ls[i].addr > tgt) break; }
		if (ti >= 50) { br = ls; bsa = ts; bti = ti; break; }
		if (ti > bti) { br = ls; bsa = ts; bti = ti; }
	}
	if (!br.empty()) { g_disPreBuffer = br; g_disPreBufferStart = bsa; g_disPreBufferEnd = br.back().addr + br.back().len; g_disPreBufferValid = true; }
}

static DisLine MakePlaceholderLine(u64 addr) { DisLine l; l.addr = addr; l.len = 1; l.bytes[0] = 0; l.mn = "??"; l.op = ""; l.isCall = l.isJmp = l.isCond = l.isRet = false; l.target = 0; l.hasMemTarget = false; l.memTarget = 0; return l; }

static std::vector<DisLine> GetDisasmLinesWithPreBuffer(u64 va, int lc) {
	auto& c = Core::I();
	if (!c.loaded || !c.mem.InRange(va)) { std::vector<DisLine> r; for (int i = 0; i < lc; i++) r.push_back(MakePlaceholderLine(va + i)); return r; }
	if (!g_disPreBufferValid || g_disPreBuffer.empty()) BuildDisasmPreBuffer(va);
	if (g_disPreBufferValid && !g_disPreBuffer.empty()) { int vi = -1; for (size_t i = 0; i < g_disPreBuffer.size(); i++) if (g_disPreBuffer[i].addr == va) { vi = (int)i; break; } if (vi >= 0 && vi < 20) BuildDisasmPreBuffer(va, true); }
	int vi = -1;
	for (size_t i = 0; i < g_disPreBuffer.size(); i++) { if (g_disPreBuffer[i].addr == va) { vi = (int)i; break; } if (g_disPreBuffer[i].addr > va) { BuildDisasmPreBuffer(va, true); for (size_t j = 0; j < g_disPreBuffer.size(); j++) if (g_disPreBuffer[j].addr == va) { vi = (int)j; break; } break; } }
	if (vi >= 0) {
		std::vector<DisLine> r; for (int i = vi; i < (int)g_disPreBuffer.size() && (int)r.size() < lc; i++) r.push_back(g_disPreBuffer[i]);
		if ((int)r.size() < lc && !r.empty()) { auto m = c.Dis(r.back().addr + r.back().len, lc - (int)r.size()); r.insert(r.end(), m.begin(), m.end()); }
		return r;
	}
	return c.Dis(va, lc);
}

static u64 GetPreviousInstructionAddress(u64 ca) {
	if (ca == 0) return 0xFFFFFFFFFFFFFFFFULL;
	auto& c = Core::I(); if (!c.loaded || !c.mem.InRange(ca)) return ca - 1;
	if (g_disPreBufferValid && !g_disPreBuffer.empty()) for (size_t i = 1; i < g_disPreBuffer.size(); i++) if (g_disPreBuffer[i].addr == ca) return g_disPreBuffer[i - 1].addr;
	BuildDisasmPreBuffer(ca);
	if (g_disPreBufferValid && !g_disPreBuffer.empty()) for (size_t i = 1; i < g_disPreBuffer.size(); i++) if (g_disPreBuffer[i].addr == ca) return g_disPreBuffer[i - 1].addr;
	for (int i = 1; i <= 15; i++) { if (ca < (u64)i) break; auto ls = c.Dis(ca - i, 2); if (!ls.empty() && ls[0].addr + ls[0].len == ca) return ls[0].addr; }
	return ca - 1;
}

static bool IsKeyPressedManual(ImGuiKey k, bool rep = true) {
	auto now = std::chrono::steady_clock::now();
	if (ImGui::IsKeyPressed(k, false)) { g_lastKey = k; g_keyDownTime = now; g_keyRepeating = false; return true; }
	if (rep && ImGui::IsKeyDown(k) && g_lastKey == k) {
		auto el = std::chrono::duration_cast<std::chrono::milliseconds>(now - g_keyDownTime).count();
		if (!g_keyRepeating) { if (el >= KEY_REPEAT_DELAY_MS) { g_keyRepeating = true; g_keyDownTime = now; return true; } }
		else { if (el >= KEY_REPEAT_RATE_MS) { g_keyDownTime = now; return true; } }
	}
	if (!ImGui::IsKeyDown(k) && g_lastKey == k) { g_lastKey = ImGuiKey_None; g_keyRepeating = false; }
	return false;
}

static bool IsValidStringChar(u8 c) { return (c >= 32 && c < 127) || c == '\t' || c == '\n' || c == '\r'; }

	return r.length() >= 4 ? r : "";
}

// === PARSE ADDRESS ===
static bool LooksLikeHex(const std::string& s) { if (s.empty()) return false; for (char c : s) if (!isxdigit(c)) return false; for (char c : s) if ((c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F')) return true; return true; }

struct ParsedAddress { u64 address; bool valid; std::string originalInput; };

static ParsedAddress ParseAddressInput(const std::string& inp) {
	ParsedAddress r = { 0,false,inp };
	auto& c = Core::I();

	std::string s = inp;
	while (!s.empty() && (s[0] == ' ' || s[0] == '\t')) s.erase(0, 1);
	while (!s.empty() && (s.back() == ' ' || s.back() == '\t')) s.pop_back();
	if (s.empty()) return r;

	// 0x prefix ile başlıyorsa kesinlikle hex
	if (s.size() >= 2 && s[0] == '0' && (s[1] == 'x' || s[1] == 'X')) {
		r.address = strtoull(s.c_str(), nullptr, 16);
		r.valid = true;
		return r;
	}

	// + veya - içeriyorsa expression olarak parse et
	size_t fp = s.find('+'), fm = s.find('-'), fo = std::string::npos;
	if (fp != std::string::npos && fm != std::string::npos) fo = std::min(fp, fm);
	else if (fp != std::string::npos) fo = fp;
	else if (fm != std::string::npos) fo = fm;

	if (fo != std::string::npos) {
		std::string bp = s.substr(0, fo), op = s.substr(fo);
		while (!bp.empty() && bp.back() == ' ') bp.pop_back();
		u64 ba = 0; bool bf = false;
		std::string lb = bp, lf = g_loadedFileName;
		for (auto& x : lb) x = tolower(x);
		for (auto& x : lf) x = tolower(x);
		if (lb == lf || lb + ".exe" == lf || lb + ".dll" == lf) { ba = c.Base(); bf = true; }
		if (!bf) { ba = c.syms.Lookup(bp); if (ba) bf = true; }
		if (bf) {
			i64 to = 0; size_t pos = 0;
			while (pos < op.size()) {
				char o = op[pos]; if (o != '+' && o != '-') break; pos++;
				while (pos < op.size() && op[pos] == ' ') pos++;
				size_t ns = pos; bool hx = false;
				if (pos + 1 < op.size() && op[pos] == '0' && (op[pos + 1] == 'x' || op[pos + 1] == 'X')) { hx = true; pos += 2; }
				while (pos < op.size() && isxdigit(op[pos])) pos++;
				std::string num = hx ? op.substr(ns, pos - ns) : op.substr(ns, pos - ns);
				while (!num.empty() && isspace(num.back())) num.pop_back();
				if (!num.empty()) {
					// Offset her zaman hex olarak parse et
					i64 v = (i64)strtoull(num.c_str() + (hx ? 2 : 0), nullptr, 16);
					if (o == '+') to += v; else to -= v;
				}
				while (pos < op.size() && op[pos] != '+' && op[pos] != '-') pos++;
			}
			r.address = (u64)((i64)ba + to); r.valid = true; return r;
		}
	}

	// Modül ismi kontrolü
	{
		std::string l = s, lf = g_loadedFileName;
		for (auto& x : l) x = tolower(x);
		for (auto& x : lf) x = tolower(x);
		if (l == lf || l + ".exe" == lf || l + ".dll" == lf) {
			r.address = c.Base(); r.valid = true; return r;
		}
	}

	// Sembol kontrolü
	u64 sa = c.syms.Lookup(s);
	if (sa) { r.address = sa; r.valid = true; return r; }

	// Import kontrolü
	{
		std::string l = s; for (auto& x : l) x = tolower(x);
		for (auto& d : c.pe.imps) {
			std::string dl = d.name; for (auto& x : dl) x = tolower(x);
			std::string dn = dl;
			if (dn.size() > 4 && dn.substr(dn.size() - 4) == ".dll") dn = dn.substr(0, dn.size() - 4);
			if ((l == dl || l == dn) && !d.funcs.empty()) {
				r.address = d.funcs[0].iatAddr; r.valid = true; return r;
			}
		}
	}

	// ÖNEMLİ: Sadece hex karakterlerden oluşuyorsa hex olarak parse et
	// Bu 4f9f00 gibi adreslerin çalışmasını sağlar
	bool allHex = true;
	bool hasHexLetter = false;
	for (char ch : s) {
		if (!isxdigit(ch)) { allHex = false; break; }
		if ((ch >= 'a' && ch <= 'f') || (ch >= 'A' && ch <= 'F')) hasHexLetter = true;
	}

	if (allHex && !s.empty()) {
		// Hex olarak parse et
		r.address = strtoull(s.c_str(), nullptr, 16);
		r.valid = true;
		return r;
	}

	// Son çare: normal parse
	char* ep = nullptr;
	u64 nv = strtoull(s.c_str(), &ep, 0);
	if (ep != s.c_str() && (*ep == '\0' || *ep == ' ')) {
		r.address = nv; r.valid = true;
	}
	return r;
}
// === NAV SEARCH ===
static void DoNavSearch() {
	g_navSearchResults.clear(); g_navSearchSelected.clear(); g_navSearchLastClicked = -1;
	auto& c = Core::I();
	if (!c.loaded || !c.refs.built) { Log("Build refs first (Ctrl+Shift+R)"); return; }
	std::string query = g_navSearchBuf;
	while (!query.empty() && (query[0] == ' ' || query[0] == '\t')) query.erase(0, 1);
	while (!query.empty() && (query.back() == ' ' || query.back() == '\t')) query.pop_back();
	if (query.empty()) return;
	std::string qLower = query; for (auto& ch : qLower) ch = tolower(ch);
	std::string instrType, addrPart;
	if (qLower.substr(0, 5) == "call ") { instrType = "call"; addrPart = query.substr(5); }
	else if (qLower.substr(0, 4) == "jmp ") { instrType = "jmp"; addrPart = query.substr(4); }
	else if (qLower.substr(0, 4) == "jcc ") { instrType = "jcc"; addrPart = query.substr(4); }
	else if (qLower.substr(0, 2) == "je " || qLower.substr(0, 3) == "jne" || qLower.substr(0, 2) == "jz " || qLower.substr(0, 3) == "jnz") { instrType = "jcc"; addrPart = query.substr(query.find(' ') + 1); }
	else { addrPart = query; }
	while (!addrPart.empty() && addrPart[0] == ' ') addrPart.erase(0, 1);
	ParsedAddress pa = ParseAddressInput(addrPart);
	if (!pa.valid) { Log("Invalid address: %s", addrPart.c_str()); return; }
	u64 targetAddr = pa.address;
	auto refs = c.refs.FindTo(targetAddr);
	for (auto& ref : refs) {
		bool match = instrType.empty();
		if (!match) {
			if (instrType == "call" && ref.type == RefInfo::CALL) match = true;
			else if (instrType == "jmp" && ref.type == RefInfo::JMP) match = true;
			else if (instrType == "jcc" && ref.type == RefInfo::JCC) match = true;
		}
		if (match) {
			SearchResult sr; sr.srcAddr = ref.src; sr.type = RefInfo::TypeStr(ref.type);
			auto lines = c.Dis(ref.src, 1);
			if (!lines.empty()) {
				auto& l = lines[0];
				std::string formattedAddr = g_showModuleNames ? FormatAddressWithModule(l.addr, true) : FormatDisplayAddress(l.addr, c.pe.is64);
				std::string formattedOp = g_showModuleNames ? FormatOperandWithModule(l.op, l.target, l.memTarget) : l.op;
				sr.instruction = formattedAddr + " - " + l.mn + " " + formattedOp;
			}
			else { char buf[64]; sprintf(buf, g_showHexPrefix ? "0x%llX" : "%llX", ref.src); sr.instruction = buf; }
			g_navSearchResults.push_back(sr);
		}
	}
	Log("Found %zu references to %s", g_navSearchResults.size(), addrPart.c_str());
}

	return DefWindowProcA(h, m, w, l);
}

	if (ImGui::Begin("Code Injection", &g_injDlgOpen, ImGuiWindowFlags_NoCollapse)) {
		if (!c.loaded) {
			ImGui::TextColored(ImVec4(1, 0.5f, 0.5f, 1), "No file loaded!");
			ImGui::End();
			return;
		}

		// ===== MODE SECTION =====
		ImGui::TextColored(ImVec4(0.9f, 0.6f, 0.3f, 1), "Injection Mode");
		ImGui::Separator();

		ImGui::RadioButton("Smart (INT3 Padding)", &g_injMode, 0);
		ImGui::SameLine();
		ImGui::TextColored(ImVec4(0.5f, 0.5f, 0.5f, 1), "- Uses INT3/NOP blocks as buffer");

		ImGui::RadioButton("Hook (Detour)", &g_injMode, 1);
		ImGui::SameLine();
		ImGui::TextColored(ImVec4(0.5f, 0.5f, 0.5f, 1), "- JMP to new location");

		// Smart mode seçenekleri
		if (g_injMode == 0) {
			ImGui::Indent(20);
			ImGui::Spacing();

			ImGui::TextColored(ImVec4(0.7f, 0.7f, 0.7f, 1), "Smart Mode Options:");

			ImGui::RadioButton("Consuming", &g_injSmartMode, 0);
			if (ImGui::IsItemHovered()) {
				ImGui::SetTooltip("Consumes INT3 bytes, only shifts code.\nMore efficient, compact result.");
			}
			ImGui::SameLine();
			ImGui::RadioButton("Full Shift", &g_injSmartMode, 1);
			if (ImGui::IsItemHovered()) {
				ImGui::SetTooltip("Shifts everything including INT3 blocks.\nPreserves original structure.");
			}

			ImGui::Checkbox("Keep 1 INT3 boundary between regions", &g_injKeepBoundary);
			if (ImGui::IsItemHovered()) {
				ImGui::SetTooltip("Keeps 1 INT3 as separator between code regions.\nHelps identify function boundaries.");
			}

			ImGui::Unindent(20);
		}

		// Hook mode seçenekleri
		if (g_injMode == 1) {
			ImGui::Indent(20);
			ImGui::Spacing();

			ImGui::TextColored(ImVec4(0.7f, 0.7f, 0.7f, 1), "Hook Target Location:");

			ImGui::RadioButton("Auto (.hook section)", &g_injTargetMode, 0);
			if (ImGui::IsItemHovered()) {
				ImGui::SetTooltip("Automatically create/use .hook section.\nSafest and recommended option.");
			}

			ImGui::RadioButton("Code Cave (find free space)", &g_injTargetMode, 1);
			if (ImGui::IsItemHovered()) {
				ImGui::SetTooltip("Find existing INT3/NOP padding in executable sections.\nNo new section needed.");
			}

			ImGui::RadioButton("Manual Address", &g_injTargetMode, 2);
			if (ImGui::IsItemHovered()) {
				ImGui::SetTooltip("Specify exact address for hook code.\nUse at your own risk!");
			}

			// Code Cave options
			if (g_injTargetMode == 1) {
				ImGui::Indent(20);
				ImGui::Spacing();

				ImGui::Text("Minimum cave size:");
				ImGui::SameLine();
				ImGui::SetNextItemWidth(80);
				ImGui::InputText("##cavesz", g_injCaveSizeBuf, sizeof(g_injCaveSizeBuf));
				ImGui::SameLine();
				if (ImGui::SmallButton("32")) strcpy(g_injCaveSizeBuf, "32");
				ImGui::SameLine();
				if (ImGui::SmallButton("64")) strcpy(g_injCaveSizeBuf, "64");
				ImGui::SameLine();
				if (ImGui::SmallButton("128")) strcpy(g_injCaveSizeBuf, "128");

				ImGui::Spacing();
				ImGui::TextColored(ImVec4(0.6f, 0.8f, 0.6f, 1), "Available Code Caves:");

				size_t minSize = (size_t)atoi(g_injCaveSizeBuf);
				if (minSize < 16) minSize = 16;

				struct CaveInfo { u64 addr; size_t size; std::string section; };
				std::vector<CaveInfo> caves;

				for (auto& sec : c.pe.secs) {
					if (!(sec.chars & 0x20000000)) continue;

					u64 secStart = c.Base() + sec.va;
					u64 secEnd = secStart + sec.vSz;

					size_t currentCave = 0;
					u64 currentCaveStart = 0;

					for (u64 addr = secStart; addr < secEnd; addr++) {
						if (c.mem.InRange(addr)) {
							u8 b = c.mem.Read8(addr);
							if (b == 0xCC || b == 0x90) {
								if (currentCave == 0) currentCaveStart = addr;
								currentCave++;
							}
							else {
								if (currentCave >= minSize) {
									CaveInfo ci;
									ci.addr = currentCaveStart;
									ci.size = currentCave;
									ci.section = sec.name;
									caves.push_back(ci);
								}
								currentCave = 0;
							}
						}
					}
					if (currentCave >= minSize) {
						CaveInfo ci;
						ci.addr = currentCaveStart;
						ci.size = currentCave;
						ci.section = sec.name;
						caves.push_back(ci);
					}
				}

				// Sort by size descending
				std::sort(caves.begin(), caves.end(), [](const CaveInfo& a, const CaveInfo& b) {
					return a.size > b.size;
					});

				if (caves.empty()) {
					ImGui::TextColored(ImVec4(1, 0.5f, 0.5f, 1), "No suitable caves found (min %zu bytes)!", minSize);
				}
				else {
					ImGui::Text("Found %zu caves:", caves.size());
					ImGui::BeginChild("##cavelist", ImVec2(0, 120), true);
					for (size_t i = 0; i < caves.size(); i++) {
						auto& cv = caves[i];
						char buf[128];
						sprintf(buf, "%s%llX - %zu bytes [%s]%s",
							g_showHexPrefix ? "0x" : "", cv.addr, cv.size, cv.section.c_str(),
							i == 0 ? " (largest)" : "");

						bool selected = false;
						if (strlen(g_injManualAddrBuf) > 0) {
							u64 manAddr = strtoull(g_injManualAddrBuf, nullptr, 16);
							selected = (manAddr == cv.addr);
						}

						if (ImGui::Selectable(buf, selected)) {
							sprintf(g_injManualAddrBuf, "%llX", cv.addr);
							g_injTargetMode = 2;
						}

						if (ImGui::IsItemHovered()) {
							ImGui::BeginTooltip();
							ImGui::Text("Click to use this cave");
							ImGui::Text("Address: %s%llX - %s%llX",
								g_showHexPrefix ? "0x" : "", cv.addr,
								g_showHexPrefix ? "0x" : "", cv.addr + cv.size - 1);
							ImGui::EndTooltip();
						}
					}
					ImGui::EndChild();
				}

				ImGui::Unindent(20);
			}

			// Manual address input
			if (g_injTargetMode == 2) {
				ImGui::Indent(20);
				ImGui::Spacing();

				ImGui::Text("Hook code address:");
				ImGui::SameLine();
				ImGui::SetNextItemWidth(150);
				ImGui::InputText("##manaddr", g_injManualAddrBuf, sizeof(g_injManualAddrBuf),
					ImGuiInputTextFlags_CharsHexadecimal);

				if (strlen(g_injManualAddrBuf) > 0) {
					u64 manAddr = strtoull(g_injManualAddrBuf, nullptr, 16);
					if (c.mem.InRange(manAddr)) {
						// Check free space
						int freeBytes = 0;
						for (int i = 0; i < 256 && c.mem.InRange(manAddr + i); i++) {
							u8 b = c.mem.Read8(manAddr + i);
							if (b == 0x00 || b == 0xCC || b == 0x90) freeBytes++;
							else break;
						}

						if (freeBytes >= 32) {
							ImGui::TextColored(ImVec4(0.5f, 1, 0.5f, 1),
								"Valid! Free space: %d bytes", freeBytes);
						}
						else if (freeBytes > 0) {
							ImGui::TextColored(ImVec4(1, 1, 0.5f, 1),
								"Warning: Only %d free bytes", freeBytes);
						}
						else {
							ImGui::TextColored(ImVec4(1, 0.5f, 0.5f, 1),
								"Warning: No free space detected!");
						}

						// Byte preview
						ImGui::Text("Bytes:");
						ImGui::SameLine();
						for (int i = 0; i < 16 && c.mem.InRange(manAddr + i); i++) {
							u8 b = c.mem.Read8(manAddr + i);
							if (b == 0xCC) ImGui::TextColored(ImVec4(0.5f, 0.5f, 0.5f, 1), "%02X", b);
							else if (b == 0x90) ImGui::TextColored(ImVec4(0.5f, 0.8f, 0.5f, 1), "%02X", b);
							else if (b == 0x00) ImGui::TextColored(ImVec4(0.3f, 0.3f, 0.5f, 1), "%02X", b);
							else ImGui::TextColored(ImVec4(1, 0.8f, 0.5f, 1), "%02X", b);
							if (i < 15) ImGui::SameLine();
						}

						// Find section
						for (auto& sec : c.pe.secs) {
							u64 secStart = c.Base() + sec.va;
							u64 secEnd = secStart + sec.vSz;
							if (manAddr >= secStart && manAddr < secEnd) {
								bool isExec = (sec.chars & 0x20000000) != 0;
								bool isWrite = (sec.chars & 0x80000000) != 0;
								ImGui::Text("Section: %s (%s%s)", sec.name,
									isExec ? "X" : "", isWrite ? "W" : "R");
								break;
							}
						}
					}
					else {
						ImGui::TextColored(ImVec4(1, 0.5f, 0.5f, 1), "Address not in memory range!");
					}
				}
				ImGui::Unindent(20);
			}

			ImGui::Unindent(20);
		}

		ImGui::Spacing();

		// ===== STRING STORAGE SECTION =====
		ImGui::TextColored(ImVec4(0.9f, 0.6f, 0.3f, 1), "String Storage");
		ImGui::Separator();

		ImGui::RadioButton(".hook section", &g_stringStorageMode, 0);
		if (ImGui::IsItemHovered()) {
			ImGui::SetTooltip("Strings written to .hook section");
		}
		ImGui::SameLine();
		ImGui::RadioButton(".text section", &g_stringStorageMode, 1);
		if (ImGui::IsItemHovered()) {
			ImGui::SetTooltip("Strings written to .text section");
		}

		g_stringStorage.currentMode = (g_stringStorageMode == 0) ?
			StringStorage::StorageMode::HookSection :
			StringStorage::StorageMode::TextSection;

		if (!g_stringStorage.entries.empty()) {
			ImGui::Spacing();
			int existingCount = 0, newCount = 0;
			for (auto& e : g_stringStorage.entries) {
				if (e.valid) {
					if (e.wasExisting) existingCount++;
					else newCount++;
				}
			}

			char headerText[64];
			sprintf(headerText, "Strings (%d existing, %d new)", existingCount, newCount);

			if (ImGui::TreeNode(headerText)) {
				if (existingCount > 0) {
					ImGui::TextColored(ImVec4(0.5f, 0.8f, 0.5f, 1), "From file:");
					for (size_t i = 0; i < g_stringStorage.entries.size(); i++) {
						auto& e = g_stringStorage.entries[i];
						if (!e.valid || !e.wasExisting) continue;
						const char* modeStr = (e.mode == StringStorage::StorageMode::HookSection) ? ".hook" : ".text";
						ImGui::PushID((int)(i + 1000));
						ImGui::BulletText("%llX [%s] \"%s\"", e.addr, modeStr, e.content.c_str());
						ImGui::SameLine();
						if (ImGui::SmallButton("Go")) {
							g_viewAddr = e.addr;
							g_selAddr = e.addr;
							g_disPreBufferValid = false;
						}
						ImGui::PopID();
					}
				}
				if (newCount > 0) {
					ImGui::Spacing();
					ImGui::TextColored(ImVec4(0.8f, 0.8f, 0.5f, 1), "Added this session:");
					for (size_t i = 0; i < g_stringStorage.entries.size(); i++) {
						auto& e = g_stringStorage.entries[i];
						if (!e.valid || e.wasExisting) continue;
						const char* modeStr = (e.mode == StringStorage::StorageMode::HookSection) ? ".hook" : ".text";
						ImGui::PushID((int)i);
						ImGui::BulletText("%llX [%s] \"%s\"", e.addr, modeStr, e.content.c_str());
						ImGui::SameLine();
						if (ImGui::SmallButton("Go")) {
							g_viewAddr = e.addr;
							g_selAddr = e.addr;
							g_disPreBufferValid = false;
						}
						ImGui::PopID();
					}
				}
				ImGui::Spacing();
				if (ImGui::Button("Rescan Sections")) {
					g_stringStorage.ScanExistingStrings(Core::I().pe, Core::I().mem);
					Log("Rescanned sections, found %zu strings", g_stringStorage.entries.size());
				}
				ImGui::TreePop();
			}
		}

		ImGui::Spacing();
		ImGui::TextColored(ImVec4(0.6f, 0.6f, 0.6f, 1), "Use @STR:text or @STR:\"text with spaces\"");

		// ===== ADDRESS SECTION =====
		ImGui::Spacing();
		ImGui::TextColored(ImVec4(0.9f, 0.6f, 0.3f, 1), "Injection Point");
		ImGui::Separator();

		ImGui::Text("Hook/Inject at:"); ImGui::SameLine();
		ImGui::SetNextItemWidth(180);
		ImGui::InputText("##ia", g_injAddrBuf, sizeof(g_injAddrBuf));
		ImGui::SameLine();
		if (ImGui::Button("Selection")) sprintf(g_injAddrBuf, "%llX", g_selAddr);

		ParsedAddress pa = ParseAddressInput(g_injAddrBuf);
		if (pa.valid && c.mem.InRange(pa.address)) {
			auto lines = c.Dis(pa.address, 3);
			if (!lines.empty()) {
				ImGui::Text("Instructions at injection point:");
				for (size_t i = 0; i < std::min(lines.size(), (size_t)3); i++) {
					ImGui::TextColored(ImVec4(0.6f, 0.6f, 0.6f, 1),
						"  %llX: %s %s", lines[i].addr, lines[i].mn.c_str(), lines[i].op.c_str());
				}
			}

			// Mode specific info
			if (g_injMode == 0) {
				// Smart mode preview
				size_t previewSize = 0;
				{
					u64 addr = pa.address;
					std::istringstream iss(g_injCodeBuf);
					std::string line;
					c.SetupAssemblerCallbacks();
					while (std::getline(iss, line)) {
						while (!line.empty() && (line[0] == ' ' || line[0] == '\t')) line.erase(0, 1);
						while (!line.empty() && (line.back() == ' ' || line.back() == '\t' || line.back() == '\r')) line.pop_back();
						if (line.empty()) continue;
						auto bytes = c.asmb.Asm(line, addr);
						previewSize += bytes.size();
						addr += bytes.size();
					}
				}

				if (previewSize > 0) {
					g_injector.smartMode = (g_injSmartMode == 0) ?
						ByteInjector::SmartMode::Consuming : ByteInjector::SmartMode::Shift;
					g_injector.keepBoundaryInt3 = g_injKeepBoundary;

					ByteInjector::ShiftPlan plan;
					if (g_injSmartMode == 0) {
						plan = g_injector.CreateConsumingPlan(c.mem, c.dis, pa.address, previewSize);
					}
					else {
						plan = g_injector.CreateShiftPlan(c.mem, c.dis, pa.address, previewSize);
					}

					if (plan.valid) {
						ImGui::TextColored(ImVec4(0.5f, 1, 0.5f, 1),
							"Needed: %d bytes | Found: %d bytes | OK!",
							plan.int3sNeeded, plan.usableInt3Bytes);
					}
					else {
						ImGui::TextColored(ImVec4(1, 0.5f, 0.5f, 1),
							"Needed: %d bytes | Found: %d bytes | NOT ENOUGH!",
							plan.int3sNeeded, plan.usableInt3Bytes);
						ImGui::TextColored(ImVec4(1, 1, 0.5f, 1), "Use Hook mode instead.");
					}
				}
			}
			else {
				// Hook mode info
				ImGui::Spacing();
				if (g_injTargetMode == 0) {
					ImGui::TextColored(ImVec4(0.5f, 0.8f, 1, 1),
						"Target: .hook section (will be created/used automatically)");
				}
				else if (g_injTargetMode == 1) {
					ImGui::TextColored(ImVec4(0.5f, 1, 0.8f, 1),
						"Target: Code Cave (largest available will be selected)");
				}
				else if (g_injTargetMode == 2 && strlen(g_injManualAddrBuf) > 0) {
					u64 manAddr = strtoull(g_injManualAddrBuf, nullptr, 16);
					ImGui::TextColored(ImVec4(1, 0.8f, 0.5f, 1),
						"Target: %s%llX (manual)", g_showHexPrefix ? "0x" : "", manAddr);
				}
			}

			// Refs status
			if (c.refs.built) {
				ImGui::TextColored(ImVec4(0.5f, 1, 0.5f, 1), "Refs: Built (%zu)", c.refs.refs.size());
			}
			else {
				ImGui::TextColored(ImVec4(1, 1, 0.5f, 1), "Refs: Not built (will auto-build for Smart mode)");
			}
		}

		ImGui::Spacing();

		// ===== CODE SECTION =====
		ImGui::TextColored(ImVec4(0.9f, 0.6f, 0.3f, 1), "Code");
		ImGui::Separator();

		ImGui::Text("Assembly (one per line):");
		ImGui::InputTextMultiline("##ic", g_injCodeBuf, sizeof(g_injCodeBuf),
			ImVec2(-1, 120), ImGuiInputTextFlags_AllowTabInput);

		// Templates
		ImGui::Text("Templates:");
		ImGui::SameLine();
		if (ImGui::SmallButton("MsgBox")) {
			strcpy(g_injCodeBuf, "push 0\npush 0\npush 0\npush 0\ncall USER32.MessageBoxA");
		}
		ImGui::SameLine();
		if (ImGui::SmallButton("MsgBox+STR")) {
			strcpy(g_injCodeBuf,
				"push 0x40\n"
				"push @STR:\"Hello Title\"\n"
				"push @STR:\"Hello Message!\"\n"
				"push 0\n"
				"call USER32.MessageBoxA");
		}
		ImGui::SameLine();
		if (ImGui::SmallButton("LoadLib+STR")) {
			strcpy(g_injCodeBuf, "push @STR:user32.dll\ncall KERNEL32.LoadLibraryA");
		}
		ImGui::SameLine();
		if (ImGui::SmallButton("NOPs")) strcpy(g_injCodeBuf, "nop\nnop\nnop\nnop\nnop");
		ImGui::SameLine();
		if (ImGui::SmallButton("INT3")) strcpy(g_injCodeBuf, "int3");
		ImGui::SameLine();
		if (ImGui::SmallButton("SaveRegs")) {
			strcpy(g_injCodeBuf, "pushad\npushfd\n; Your code here\npopfd\npopad");
		}

		ImGui::Spacing();
		ImGui::Separator();

		// ===== ACTIONS =====
		bool canDo = pa.valid && c.mem.InRange(pa.address) && strlen(g_injCodeBuf) > 0;

		// Hook mode + manual/cave address validation
		if (g_injMode == 1 && g_injTargetMode == 2) {
			if (strlen(g_injManualAddrBuf) == 0) canDo = false;
			else {
				u64 manAddr = strtoull(g_injManualAddrBuf, nullptr, 16);
				if (!c.mem.InRange(manAddr)) canDo = false;
			}
		}

		if (!canDo) ImGui::BeginDisabled();

		if (ImGui::Button("Preview", ImVec2(90, 0))) {
			g_injLog = "╔════════════════════════════════════════╗\n";
			g_injLog += "║  PREVIEW   ║\n";
			g_injLog += "╚════════════════════════════════════════╝\n\n";
			c.SetupAssemblerCallbacks();

			u64 addr = pa.address;
			std::istringstream iss(g_injCodeBuf);
			std::string line;
			size_t total = 0;

			while (std::getline(iss, line)) {
				while (!line.empty() && (line[0] == ' ' || line[0] == '\t')) line.erase(0, 1);
				while (!line.empty() && (line.back() == ' ' || line.back() == '\t' || line.back() == '\r')) line.pop_back();
				if (line.empty()) continue;

				auto bytes = c.asmb.Asm(line, addr);
				if (!bytes.empty()) {
					char buf[256];
					sprintf(buf, "%llX: ", addr);
					g_injLog += buf;
					for (u8 b : bytes) { sprintf(buf, "%02X ", b); g_injLog += buf; }
					g_injLog += "; " + line + "\n";
					total += bytes.size();
					addr += bytes.size();
				}
				else {
					g_injLog += "ERROR: Cannot assemble: " + line + "\n";
				}
			}

			char buf[256];
			sprintf(buf, "\nTotal: %zu bytes\n", total);
			g_injLog += buf;

			// Mode info
			if (g_injMode == 0) {
				g_injLog += "\nMode: Smart (";
				g_injLog += (g_injSmartMode == 0) ? "Consuming" : "Full Shift";
				g_injLog += ")\n";
			}
			else {
				g_injLog += "\nMode: Hook (Detour)\n";
				if (g_injTargetMode == 0) g_injLog += "Target: .hook section (auto)\n";
				else if (g_injTargetMode == 1) g_injLog += "Target: Code Cave (auto-select)\n";
				else {
					sprintf(buf, "Target: Manual (%s%s)\n", g_showHexPrefix ? "0x" : "", g_injManualAddrBuf);
					g_injLog += buf;
				}
			}
		}

		ImGui::SameLine();

		if (ImGui::Button("Inject!", ImVec2(90, 0))) {
			g_stringStorage.currentMode = (g_stringStorageMode == 0) ?
				StringStorage::StorageMode::HookSection :
				StringStorage::StorageMode::TextSection;

			c.SetupAssemblerCallbacks();
			ByteInjector::Result res;

			g_injector.smartMode = (g_injSmartMode == 0) ?
				ByteInjector::SmartMode::Consuming : ByteInjector::SmartMode::Shift;
			g_injector.keepBoundaryInt3 = g_injKeepBoundary;

			if (g_injMode == 0) {
				// Smart mode
				std::vector<u8> bytes;
				u64 addr = pa.address;
				std::istringstream iss(g_injCodeBuf);
				std::string line;
				bool asmOk = true;

				while (std::getline(iss, line) && asmOk) {
					while (!line.empty() && (line[0] == ' ' || line[0] == '\t')) line.erase(0, 1);
					while (!line.empty() && (line.back() == ' ' || line.back() == '\t' || line.back() == '\r')) line.pop_back();
					if (line.empty()) continue;

					auto lb = c.asmb.Asm(line, addr);
					if (lb.empty()) {
						asmOk = false;
						g_injLog = "Assemble failed: " + line + "\n";
					}
					else {
						bytes.insert(bytes.end(), lb.begin(), lb.end());
						addr += lb.size();
					}
				}

				if (asmOk && !bytes.empty()) {
					res = g_injector.SmartInject(c.pe, c.mem, c.dis, c.refs, pa.address, bytes);
				}
			}
			else {
				// Hook mode
				if (g_injTargetMode == 0) {
					// Auto - .hook section
					res = g_injector.CreateHook(c.pe, c.mem, c.dis, c.asmb, pa.address, g_injCodeBuf);
				}
				else if (g_injTargetMode == 2 && strlen(g_injManualAddrBuf) > 0) {
					// Manual address
					u64 targetAddr = strtoull(g_injManualAddrBuf, nullptr, 16);
					res = g_injector.CreateHookAt(c.pe, c.mem, c.dis, c.asmb,
						pa.address, g_injCodeBuf, targetAddr);
				}
				else if (g_injTargetMode == 1) {
					// Code Cave - find largest
					u64 targetAddr = 0;
					size_t largestSize = 0;

					for (auto& sec : c.pe.secs) {
						if (!(sec.chars & 0x20000000)) continue;

						u64 secStart = c.Base() + sec.va;
						u64 secEnd = secStart + sec.vSz;

						size_t currentCave = 0;
						u64 currentCaveStart = 0;

						for (u64 addr = secStart; addr < secEnd; addr++) {
							if (c.mem.InRange(addr)) {
								u8 b = c.mem.Read8(addr);
								if (b == 0xCC || b == 0x90) {
									if (currentCave == 0) currentCaveStart = addr;
									currentCave++;
								}
								else {
									if (currentCave > largestSize) {
										largestSize = currentCave;
										targetAddr = currentCaveStart;
									}
									currentCave = 0;
								}
							}
						}
						if (currentCave > largestSize) {
							largestSize = currentCave;
							targetAddr = currentCaveStart;
						}
					}

					if (targetAddr == 0) {
						g_injLog = "ERROR: No suitable code cave found!\n";
						res.ok = false;
						res.err = "No code cave found";
					}
					else {
						res = g_injector.CreateHookAt(c.pe, c.mem, c.dis, c.asmb,
							pa.address, g_injCodeBuf, targetAddr);
					}
				}
			}

			if (res.ok) {
				g_injLog = res.log;
				g_disPreBufferValid = false;
				c.refs.refs.clear();
				c.refs.toMap.clear();
				c.refs.fromMap.clear();
				c.refs.built = false;
				Log("Injected at %llX (%zu bytes), hook code at %llX",
					res.addr, res.size, res.hookAddr);
			}
			else {
				g_injLog = "╔════════════════════════════════════════╗\n";
				g_injLog += "║  FAILED║\n";
				g_injLog += "╚════════════════════════════════════════╝\n\n";
				g_injLog += "Error: " + res.err + "\n\n";
				g_injLog += res.log;
			}
		}

		if (!canDo) ImGui::EndDisabled();

		ImGui::SameLine();
		if (ImGui::Button("Copy Log", ImVec2(90, 0))) {
			if (!g_injLog.empty()) {
				ImGui::SetClipboardText(g_injLog.c_str());
				Log("Log copied to clipboard");
			}
		}
		ImGui::SameLine();
		if (ImGui::Button("Close", ImVec2(90, 0))) g_injDlgOpen = false;

		// ===== LOG SECTION =====
		if (!g_injLog.empty()) {
			ImGui::Spacing();
			ImGui::TextColored(ImVec4(0.9f, 0.6f, 0.3f, 1), "Log");
			ImGui::Separator();
			float lh = ImGui::GetContentRegionAvail().y - 10;
			if (lh < 100) lh = 100;
			ImGui::BeginChild("##il", ImVec2(-1, lh), true, ImGuiWindowFlags_HorizontalScrollbar);
			if (GetDisasmFont()) ImGui::PushFont(GetDisasmFont());
			ImGui::TextUnformatted(g_injLog.c_str());
			if (GetDisasmFont()) ImGui::PopFont();
			ImGui::EndChild();
		}
	}
	ImGui::End();
}

static int FindLineIndex(const std::vector<DisLine>& ls, u64 a) { for (size_t i = 0; i < ls.size(); i++) if (ls[i].addr == a) return (int)i; return -1; }

// === BRANCH LINES ===
static void CalculateBranchLines(const std::vector<DisLine>& ls, int vl) {
	g_branchLines.clear(); g_maxBranchLane = 0; if (ls.empty()) return;
	std::unordered_map<u64, int> a2i; int mx = std::min((int)ls.size(), vl);
	for (int i = 0; i < mx; i++) a2i[ls[i].addr] = i;
	std::vector<std::vector<std::pair<int, int>>> lo;
	auto findLane = [&](int y1, int y2)->int {
		int mi = std::min(y1, y2), ma = std::max(y1, y2);
		for (int l = 0; l < (int)lo.size(); l++) { bool oc = false; for (auto& r : lo[l]) if (!(ma < r.first || mi > r.second)) { oc = true; break; } if (!oc) { lo[l].push_back({ mi,ma }); return l; } }
		int nl = (int)lo.size(); lo.push_back({ {mi,ma} }); return nl;
	};
	for (int i = 0; i < mx; i++) {
		auto& l = ls[i]; if (!l.isJmp && !l.isCall && !l.isCond) continue;
		u64 tgt = l.target ? l.target : (l.hasMemTarget ? l.memTarget : 0); if (!tgt) continue;
		auto it = a2i.find(tgt); if (it == a2i.end()) continue;
		int di = it->second; if (di == i) continue;
		BranchLine bl; bl.srcInstrIdx = i; bl.dstInstrIdx = di; bl.srcAddr = l.addr; bl.dstAddr = tgt;
		bl.isCall = l.isCall; bl.isConditional = l.isCond; bl.isForward = (tgt > l.addr);
		bl.isSelected = (l.addr == g_selAddr || tgt == g_selAddr || g_selectedAddrs.count(l.addr) || g_selectedAddrs.count(tgt));
		bl.color = bl.isSelected ? g_colors.White : (l.isCall ? g_colors.Green : (l.isCond ? g_colors.Red : g_colors.Orange));
		bl.lane = findLane(i, di); if (bl.lane > g_maxBranchLane) g_maxBranchLane = bl.lane;
		g_branchLines.push_back(bl);
	}
}

	ImDrawList* dl = ImGui::GetWindowDrawList();
	ImGuiIO& io = ImGui::GetIO();
	ImVec2 wp = ImGui::GetWindowPos(), ws = ImGui::GetWindowSize();
	ImFont* fn = GetDisasmFont();
	ImFont* fb = GetDisasmFontBold();
	float fs = GetDisasmFontSize();

	// Info bar
	float ibh = 24;
	{
		ImU32 ibg = (g_themeMode == ThemeMode::Dark) ? IM_COL32(25, 30, 40, 255) : IM_COL32(220, 225, 235, 255);
		ImU32 ibd = (g_themeMode == ThemeMode::Dark) ? IM_COL32(50, 60, 80, 255) : IM_COL32(190, 195, 205, 255);
		ImVec2 imi = wp, ima(wp.x + cw - 4, wp.y + ibh);
		dl->AddRectFilled(imi, ima, ibg);
		dl->AddLine(ImVec2(imi.x, ima.y), ImVec2(ima.x, ima.y), ibd);
		std::string is = "Disasm - " + (g_showModuleNames ? FormatAddressWithModule(g_viewAddr, true) : FormatDisplayAddress(g_viewAddr, c.loaded ? c.pe.is64 : true));
		if (c.loaded && g_viewAddr >= c.Base() && g_viewAddr < c.Base() + c.Size()) {
			char mi[128];
			sprintf(mi, " | %s: %s%llX-%s%llX", g_loadedFileName.c_str(), g_showHexPrefix ? "0x" : "", g_useCustomBase ? g_displayBase : c.Base(), g_showHexPrefix ? "0x" : "", (g_useCustomBase ? g_displayBase : c.Base()) + c.Size());
			is += mi;
		}
		dl->AddText(fn, fs, ImVec2(wp.x + 8, wp.y + (ibh - fs) / 2), g_colors.Text, is.c_str());
	}

	float hh = 26, thh = ibh + hh, lh = fs + 4, ah = reg.y - thh - 8;

	// GERÇEK görünen satır sayısı
	int actualVisibleLines = std::max((int)(ah / lh), 10);
	int bufferLines = std::max(actualVisibleLines + 10, g_disVisibleLines);
	int vl = actualVisibleLines;

	g_disLines = GetDisasmLinesWithPreBuffer(g_viewAddr, bufferLines);
	auto& ls = g_disLines;
	CalculateBranchLines(ls, vl);

	float jlw = 80;
	if (g_maxBranchLane > 0) jlw = std::max(jlw, (g_maxBranchLane + 2) * 8.0f + 10);
	for (auto& col : g_columnOrder)
		if (col.type == DisasmColumn::Jump) { col.width = std::max(col.width, jlw); break; }

	RenderColumnHeaders(dl, ImVec2(wp.x, wp.y + ibh), cw - 4, hh, jlw);
	ImVec2 cs(wp.x, wp.y + thh);

	// === HELPER: Seçili adres görünür alanda mı kontrolü ===
	auto isAddrVisibleInCurrentView = [&](u64 addr) -> bool {
		for (int i = 0; i < std::min((int)ls.size(), vl); i++) {
			if (ls[i].addr == addr) return true;
		}
		return false;
	};

	// === HELPER: Seçili adresi görünür alana getir ===
	auto scrollToMakeVisible = [&](u64 addr, bool centerView = false) {
		if (!isAddrVisibleInCurrentView(addr)) {
			if (centerView) {
				// Ortaya getirmek için biraz geri git
				u64 targetView = addr;
				for (int i = 0; i < vl / 2; i++) {
					targetView = GetPreviousInstructionAddress(targetView);
				}
				g_viewAddr = targetView;
			}
			else {
				g_viewAddr = addr;
			}
			g_disPreBufferValid = false;
			g_disLines = GetDisasmLinesWithPreBuffer(g_viewAddr, bufferLines);
		}
	};

	// Keyboard handling
	if (isFoc && g_resizingColumn < 0) {
		bool sh = io.KeyShift;
		int si = FindLineIndex(ls, g_selAddr);

		if (IsKeyPressedManual(ImGuiKey_LeftArrow)) {
			if (g_viewAddr > 0) g_viewAddr--;
			else g_viewAddr = 0xFFFFFFFFFFFFFFFFULL;
			g_disPreBufferValid = false;
		}

		if (IsKeyPressedManual(ImGuiKey_RightArrow)) {
			if (g_viewAddr < 0xFFFFFFFFFFFFFFFFULL) g_viewAddr++;
			else g_viewAddr = 0;
			g_disPreBufferValid = false;
		}

		if (IsKeyPressedManual(ImGuiKey_UpArrow)) {
			u64 pa = GetPreviousInstructionAddress(g_selAddr);

			if (sh) {
				if (g_selectedAddrs.empty()) {
					g_selectedAddrs.insert(g_selAddr);
					g_selectedAddrs.insert(pa);
				}
				else {
					u64 maxAddr = *g_selectedAddrs.rbegin();
					if (maxAddr == g_selAddr && g_selectedAddrs.size() > 1)
						g_selectedAddrs.erase(maxAddr);
					else
						g_selectedAddrs.insert(pa);
				}
			}
			else g_selectedAddrs.clear();

			g_selAddr = pa;

			// DÜZELTME: Seçili adres görünür alana getir
			scrollToMakeVisible(g_selAddr, false);
			// Buffer'ı güncelle
			ls = g_disLines;
		}

		if (IsKeyPressedManual(ImGuiKey_DownArrow)) {
			// Önce seçili adresin görünür alanda olup olmadığını kontrol et
			si = FindLineIndex(ls, g_selAddr);

			if (si < 0) {
				// Seçili adres görünür alanda değil - önce görünür alana getir
				scrollToMakeVisible(g_selAddr, true);
				ls = g_disLines;
				si = FindLineIndex(ls, g_selAddr);
			}

			if (si >= 0 && si + 1 < (int)ls.size()) {
				u64 nextAddr = ls[si + 1].addr;

				if (sh) {
					if (g_selectedAddrs.empty()) {
						g_selectedAddrs.insert(g_selAddr);
						g_selectedAddrs.insert(nextAddr);
					}
					else {
						u64 minAddr = *g_selectedAddrs.begin();
						if (minAddr == g_selAddr && g_selectedAddrs.size() > 1)
							g_selectedAddrs.erase(minAddr);
						else
							g_selectedAddrs.insert(nextAddr);
					}
				}
				else g_selectedAddrs.clear();

				g_selAddr = nextAddr;

				// Seçili satır görünür alanın alt kısmına yaklaştıysa scroll et
				int newSi = FindLineIndex(ls, g_selAddr);
				if (newSi >= actualVisibleLines - 2 && ls.size() > 1) {
					g_viewAddr = ls[1].addr;
					g_disPreBufferValid = false;
					g_disLines = GetDisasmLinesWithPreBuffer(g_viewAddr, bufferLines);
					ls = g_disLines;
				}
			}
			else if (si == -1) {
				// Bu duruma artık düşmemeli çünkü yukarıda kontrol ettik
				// Ama yine de fallback olarak bırakalım
				if (!sh) g_selectedAddrs.clear();
			}
		}

		if (ImGui::IsKeyPressed(ImGuiKey_PageUp, false)) {
			for (int i = 0; i < actualVisibleLines - 2; i++)
				g_viewAddr = GetPreviousInstructionAddress(g_viewAddr);
			g_selAddr = g_viewAddr;
			g_selectedAddrs.clear();
			g_disPreBufferValid = false;
		}

		if (ImGui::IsKeyPressed(ImGuiKey_PageDown, false)) {
			if (ls.size() > (size_t)(actualVisibleLines - 2))
				g_viewAddr = ls[actualVisibleLines - 2].addr;
			g_selAddr = g_viewAddr;
			g_selectedAddrs.clear();
			g_disPreBufferValid = false;
		}

		if (ImGui::IsKeyPressed(ImGuiKey_Space, false)) {
			int i = FindLineIndex(ls, g_selAddr);
			if (i >= 0) {
				auto& l = ls[i];
				u64 t = l.target ? l.target : (l.hasMemTarget ? l.memTarget : 0);
				if (t) {
					char lb[64];
					sprintf(lb, "%s %s%llX", l.mn.c_str(), g_showHexPrefix ? "0x" : "", t);
					NavGotoWithSource(l.addr, t, lb, true, false);
				}
			}
		}

		if (ImGui::IsKeyPressed(ImGuiKey_Enter, false) || ImGui::IsKeyPressed(ImGuiKey_F2, false)) {
			int i = FindLineIndex(ls, g_selAddr);
			if (i >= 0) {
				g_editAddr = ls[i].addr;
				sprintf(g_editBuf, "%s %s", ls[i].mn.c_str(), ls[i].op.c_str());
				g_editOpen = true;
			}
		}

		if (ImGui::IsKeyPressed(ImGuiKey_Backspace, false)) NavBackDisasm();

		if (io.KeyCtrl && ImGui::IsKeyPressed(ImGuiKey_C, false)) {
			if (!g_selectedAddrs.empty()) {
				g_copyFrom = *g_selectedAddrs.begin();
				g_copyTo = *g_selectedAddrs.rbegin();
			}
			else {
				g_copyFrom = g_copyTo = g_selAddr;
			}
			sprintf(g_copyFromBuf, "%llX", g_copyFrom);
			sprintf(g_copyToBuf, "%llX", g_copyTo);
			g_copyOpen = true;
		}

		if (io.KeyCtrl && ImGui::IsKeyPressed(ImGuiKey_X, false)) {
			if (c.refs.built && c.refs.HasRefsTo(g_selAddr)) {
				g_refsPopupAddr = g_selAddr;
				g_refsPopupList = c.refs.FindTo(g_selAddr);
				g_refsPopupOpen = true;
			}
		}
	}

	// Mouse wheel - seçimi değiştirmez, sadece view'u kaydırır
	if (ImGui::IsWindowHovered() && g_resizingColumn < 0 && !dop) {
		float w = io.MouseWheel;
		if (w != 0) {
			if (w > 0)
				for (int i = 0; i < (int)(w * 3); i++)
					g_viewAddr = GetPreviousInstructionAddress(g_viewAddr);
			else {
				int sl = (int)(-w * 3);
				if (sl < (int)ls.size())
					g_viewAddr = ls[sl].addr;
			}
			g_disPreBufferValid = false;
			// g_selAddr ve g_selectedAddrs DEĞİŞMİYOR - bu doğru
		}
	}

	g_instrYPositions.clear();
	std::map<DisasmColumn, float> cx;
	float x = wp.x;
	for (auto& col : g_columnOrder) {
		if (!col.visible) continue;
		cx[col.type] = x;
		float w = col.width;
		if (col.type == DisasmColumn::Jump) w = std::max(w, jlw);
		x += w;
	}

	int rl = 0;
	float cy = cs.y;

	for (size_t li = 0; li < ls.size() && rl < vl; li++) {
		auto& l = ls[li];
		bool sel = (l.addr == g_selAddr) || g_selectedAddrs.count(l.addr);
		bool va = c.loaded ? c.mem.InRange(l.addr) : false;

		// Refs gösterimi
		if (c.loaded && c.refs.built && c.refs.HasRefsTo(l.addr)) {
			auto refs = c.refs.FindTo(l.addr);
			int mx = 3, rs = std::min((int)refs.size(), mx), ex = (int)refs.size() - mx;
			float rsx = cx.count(DisasmColumn::Address) ? cx[DisasmColumn::Address] : wp.x;

			for (int ri = 0; ri < rs && rl < vl; ri++) {
				auto& r = refs[ri];
				ImVec2 rls(rsx, cy), rle(wp.x + cw - 4, cy + lh);
				if (rl % 2 == 1) dl->AddRectFilled(rls, rle, g_colors.BackgroundAlt);
				char rb[128];
				sprintf(rb, "  <- %s(%s%llX)", RefInfo::TypeStr(r.type), g_showHexPrefix ? "0x" : "", r.src);
				dl->AddText(fn, fs, ImVec2(rsx + 20, cy + (lh - fs) / 2), g_colors.Green, rb);
				if (ImGui::IsMouseHoveringRect(rls, rle) && !dop) {
					dl->AddRectFilled(rls, rle, IM_COL32(60, 80, 60, 100));
					if (ImGui::IsMouseClicked(0)) {
						char lb[64];
						sprintf(lb, "%s->%s%llX", RefInfo::TypeStr(r.type), g_showHexPrefix ? "0x" : "", l.addr);
						NavGotoWithSwitch(r.src, lb, true, false);
						g_focusedPanel = FOCUS_DISASM;
					}
				}
				rl++; cy += lh;
			}

			if (ex > 0 && rl < vl) {
				ImVec2 mls(rsx, cy), mle(wp.x + cw - 4, cy + lh);
				char mb[64];
				sprintf(mb, "  <- ...(%d more)", ex);
				dl->AddText(fn, fs, ImVec2(rsx + 20, cy + (lh - fs) / 2), g_colors.TextMuted, mb);
				if (ImGui::IsMouseHoveringRect(mls, mle) && !dop) {
					dl->AddRectFilled(mls, mle, IM_COL32(60, 80, 60, 100));
					if (ImGui::IsMouseClicked(0)) {
						g_refsPopupAddr = l.addr;
						g_refsPopupList = refs;
						g_refsPopupOpen = true;
					}
				}
				rl++; cy += lh;
			}
			if (rl >= vl) break;
		}

		g_instrYPositions[l.addr] = cy + lh / 2;
		float lsx = cx.empty() ? wp.x : cx.begin()->second;

		if (sel) {
			ImVec2 smi(lsx, cy), sma(wp.x + cw - 4, cy + lh);
			dl->AddRectFilled(smi, sma, g_colors.Selection);
			dl->AddLine(ImVec2(lsx, cy), ImVec2(lsx, cy + lh), g_colors.SelectionBorder, 2);
		}
		else if (rl % 2 == 1) {
			dl->AddRectFilled(ImVec2(lsx, cy), ImVec2(wp.x + cw - 4, cy + lh), g_colors.BackgroundAlt);
		}

		float ty = cy + (lh - fs) / 2;

		for (auto& cc : g_columnOrder) {
			if (!cc.visible || cx.find(cc.type) == cx.end()) continue;
			float xx = cx[cc.type], ww = cc.width;
			if (cc.type == DisasmColumn::Jump) ww = std::max(ww, jlw);

			switch (cc.type) {
			case DisasmColumn::Address: {
				std::string as = g_showModuleNames && va ? FormatAddressWithModule(l.addr, true) : FormatDisplayAddress(l.addr, c.loaded ? c.pe.is64 : true);
				dl->PushClipRect(ImVec2(xx, cy), ImVec2(xx + ww - 4, cy + lh), true);
				dl->AddText(fn, fs, ImVec2(xx + 4, ty), g_colors.White, as.c_str());
				dl->PopClipRect();
				dl->AddLine(ImVec2(xx + ww, cy), ImVec2(xx + ww, cy + lh), g_colors.Border);
				break;
			}
			case DisasmColumn::Bytes: {
				dl->PushClipRect(ImVec2(xx, cy), ImVec2(xx + ww - 4, cy + lh), true);
				if (!va || l.mn == "??")
					dl->AddText(fn, fs, ImVec2(xx + 4, ty), g_colors.Gray, "??");
				else {
					char bs[40] = { 0 };
					for (int i = 0; i < l.len && i < 8; i++)
						sprintf(bs + i * 3, "%02X ", l.bytes[i]);
					if (l.len > 8) strcat(bs, "...");
					dl->AddText(fn, fs, ImVec2(xx + 4, ty), g_colors.White, bs);
				}
				dl->PopClipRect();
				dl->AddLine(ImVec2(xx + ww, cy), ImVec2(xx + ww, cy + lh), g_colors.Border);
				break;
			}
			case DisasmColumn::Jump: {
				if (va && (l.isJmp || l.isCall || l.isCond)) {
					u64 tgt = l.target ? l.target : (l.hasMemTarget ? l.memTarget : 0);
					if (tgt) {
						bool tv = false;
						for (auto& bl : g_branchLines)
							if (bl.srcAddr == l.addr) { tv = true; break; }
						if (!tv) {
							ImU32 ac = l.isCall ? g_colors.Green : (l.isCond ? g_colors.Red : g_colors.Orange);
							float ax = xx + ww - 10, ay = cy + lh / 2;
							if (tgt > l.addr)
								dl->AddTriangleFilled(ImVec2(ax, ay + 5), ImVec2(ax - 5, ay - 2), ImVec2(ax + 5, ay - 2), ac);
							else
								dl->AddTriangleFilled(ImVec2(ax, ay - 5), ImVec2(ax - 5, ay + 2), ImVec2(ax + 5, ay + 2), ac);
						}
					}
				}
				dl->AddLine(ImVec2(xx + ww, cy), ImVec2(xx + ww, cy + lh), g_colors.Border);
				break;
			}
			case DisasmColumn::Opcode: {
				dl->PushClipRect(ImVec2(xx, cy), ImVec2(xx + ww - 4, cy + lh), true);
				if (!va || l.mn == "??")
					dl->AddText(fn, fs, ImVec2(xx + 4, ty), g_colors.Gray, "??");
				else {
					float ox = xx + 4;
					ImU32 mc = g_colors.White;
					if (l.mn == "??" || l.mn == "db") mc = g_colors.Gray;
					else if (l.isCall) mc = g_colors.White;
					else if (l.isJmp && !l.isCond) mc = g_colors.Green;
					else if (l.isCond) mc = g_colors.Red;
					else if (l.isRet) mc = g_colors.Red;
					else if (l.mn == "push" || l.mn == "pop") mc = g_colors.Cyan;
					else if (l.mn == "nop") mc = g_colors.Gray;
					dl->AddText(fb, fs, ImVec2(ox, ty), mc, l.mn.c_str());
					ImVec2 msz = fb->CalcTextSizeA(fs, FLT_MAX, 0, l.mn.c_str());
					float tw = 60;
					if (l.mn.length() > 6) tw = 80;
					float opx = ox + std::max(msz.x + 8, tw);
					std::string ops = g_showModuleNames ? FormatOperandWithModule(l.op, l.target, l.memTarget) : l.op;
					if (!ops.empty()) {
						auto tks = ParseOperandTokens(ops, g_showModuleNames);
						float dx = opx;
						for (auto& tk : tks) {
							ImVec2 tsz = fn->CalcTextSizeA(fs, FLT_MAX, 0, tk.text.c_str());
							dl->AddText(fn, fs, ImVec2(dx, ty), tk.color, tk.text.c_str());
							dx += tsz.x;
						}
					}
				}
				dl->PopClipRect();
				dl->AddLine(ImVec2(xx + ww, cy), ImVec2(xx + ww, cy + lh), g_colors.Border);
				break;
			}
			case DisasmColumn::Comments: {
				if (va && l.mn != "??") {
					std::string cs2;
					if ((l.mn == "mov" || l.mn == "lea" || l.mn == "push") && (l.target || l.memTarget)) {
						std::string s = DetectStringAtAddress(c.mem, l.target ? l.target : l.memTarget);
						if (!s.empty()) {
							if (s.length() > 40) s = s.substr(0, 40) + "...";
							cs2 = "\"" + s + "\"";
						}
					}
					if (cs2.empty() && !l.sym.empty() && l.sym != l.op) cs2 = l.sym;
					if (!cs2.empty()) {
						dl->PushClipRect(ImVec2(xx, cy), ImVec2(wp.x + cw - 4, cy + lh), true);
						dl->AddText(fn, fs, ImVec2(xx + 4, ty), g_colors.White, cs2.c_str());
						dl->PopClipRect();
					}
				}
				break;
			}
			default: break;
			}
		}

		// Mouse click - SHIFT+CLICK DÜZELTME
		ImVec2 lmi(lsx, cy), lma(wp.x + cw - 4, cy + lh);
		if (ImGui::IsMouseHoveringRect(lmi, lma) && g_resizingColumn < 0 && !dop) {
			if (ImGui::IsMouseClicked(0)) {
				if (io.KeyShift && g_selAddr != l.addr) {
					// DÜZELTME: Anchor g_selAddr olarak kalır, tüm aralığı seç
					u64 anchor = g_selAddr;  // Anchor değişmez!
					u64 st = std::min(anchor, l.addr);
					u64 en = std::max(anchor, l.addr);

					g_selectedAddrs.clear();

					// DÜZELTME: Byte farkına göre yeterli satır hesapla
					// Her byte için 1 satır varsay (en kötü durum)
					size_t byteDiff = en - st;
					int estimatedLines = (int)(byteDiff)+100;
					// Makul bir üst limit koy
					estimatedLines = std::clamp(estimatedLines, 100, 100000);

					auto rangeLines = c.Dis(st, estimatedLines);
					for (auto& rl2 : rangeLines) {
						if (rl2.addr >= st && rl2.addr <= en) {
							g_selectedAddrs.insert(rl2.addr);
						}
						if (rl2.addr > en) break;
					}

					// ÖNEMLI: g_selAddr (anchor) DEĞİŞMEZ!
					// Böylece sonraki Shift+Click'ler de aynı anchor'dan başlar
				}
				else if (io.KeyCtrl) {
					// Ctrl+Click: toggle selection
					if (g_selectedAddrs.count(l.addr)) {
						g_selectedAddrs.erase(l.addr);
					}
					else {
						g_selectedAddrs.insert(l.addr);
					}
					g_selAddr = l.addr;
				}
				else {
					// Normal tıklama - seçimi temizle ve yeni anchor belirle
					g_selectedAddrs.clear();
					g_selAddr = l.addr;
				}
				g_focusedPanel = FOCUS_DISASM;
			}
			if (ImGui::IsMouseDoubleClicked(0)) {
				g_editAddr = l.addr;
				sprintf(g_editBuf, "%s %s", l.mn.c_str(), l.op.c_str());
				g_editOpen = true;
			}
		}

		rl++; cy += lh;
	}

	// Branch lines
	float jcx = 0;
	for (auto& col : g_columnOrder)
		if (col.type == DisasmColumn::Jump && col.visible) { jcx = cx[DisasmColumn::Jump]; break; }

	if (jcx > 0) {
		float ejw = jlw;
		for (auto& col : g_columnOrder)
			if (col.type == DisasmColumn::Jump) { ejw = std::max(col.width, jlw); break; }

		for (auto& bl : g_branchLines) {
			auto si = g_instrYPositions.find(bl.srcAddr), di = g_instrYPositions.find(bl.dstAddr);
			if (si == g_instrYPositions.end() || di == g_instrYPositions.end()) continue;
			float sy = si->second, dy = di->second, ls2 = 8, as2 = 4;
			float lx = jcx + ejw - (bl.lane + 1) * ls2 - 4, sx = jcx + ejw - 2, th = bl.isSelected ? 2.5f : 1.5f;
			dl->AddLine(ImVec2(sx, sy), ImVec2(lx, sy), bl.color, th);
			dl->AddLine(ImVec2(lx, sy), ImVec2(lx, dy), bl.color, th);
			dl->AddLine(ImVec2(lx, dy), ImVec2(sx, dy), bl.color, th);
			dl->AddTriangleFilled(ImVec2(sx, dy), ImVec2(sx - as2 * 2, dy - as2), ImVec2(sx - as2 * 2, dy + as2), bl.color);
			dl->AddCircleFilled(ImVec2(sx, sy), bl.isSelected ? 3 : 2, bl.color);
		}
	}

	ImGui::EndChild();
	ImGui::PopAllowKeyboardFocus();

	// Scrollbar
	ImGui::SameLine();
	ImVec2 sp = ImGui::GetCursorScreenPos();
	int sd = 0;
	RenderCustomScrollbar("##ds", sp.x, sp.y, sbw, reg.y, &sd);
	if (sd != 0 && !dop) {
		if (sd < 0)
			for (int i = 0; i < abs(sd); i++)
				g_viewAddr = GetPreviousInstructionAddress(g_viewAddr);
		else {
			int av = 4;
			if (!ls.empty()) {
				int t = 0;
				for (auto& ll : ls) t += ll.len;
				av = t / (int)ls.size();
				if (av < 1) av = 1;
			}
			g_viewAddr += sd * av;
		}
		g_disPreBufferValid = false;
	}
	ImGui::Dummy(ImVec2(sbw, reg.y));
}

static void OpenHexEditDialog(u64 addr, HexViewType type) {
	auto& c = Core::I();
	if (!c.loaded || !c.mem.InRange(addr)) return;

	g_hexEditDlgAddr = addr;
	g_hexEditDlgType = type;
	g_hexEditDlgOpen = true;

	int sz = GetHexViewSize(type);
	u8 data[8] = { 0 };
	for (int i = 0; i < sz && c.mem.InRange(addr + i); i++) {
		data[i] = c.mem.Read8(addr + i);
	}

	switch (type) {
	case HexViewType::ByteHex: sprintf(g_hexEditDlgBuf, "%02X", data[0]); break;
	case HexViewType::ByteDec: sprintf(g_hexEditDlgBuf, "%d", data[0]); break;
	case HexViewType::Word2Hex: sprintf(g_hexEditDlgBuf, "%04X", *(u16*)data); break;
	case HexViewType::Word2Dec: sprintf(g_hexEditDlgBuf, "%d", *(u16*)data); break;
	case HexViewType::Word4Hex: sprintf(g_hexEditDlgBuf, "%08X", *(u32*)data); break;
	case HexViewType::Word4Dec: sprintf(g_hexEditDlgBuf, "%u", *(u32*)data); break;
	case HexViewType::Word8Hex: sprintf(g_hexEditDlgBuf, "%016llX", *(u64*)data); break;
	case HexViewType::Word8Dec: sprintf(g_hexEditDlgBuf, "%llu", *(u64*)data); break;
	case HexViewType::Float: sprintf(g_hexEditDlgBuf, "%.15g", *(float*)data); break;
	case HexViewType::Double: sprintf(g_hexEditDlgBuf, "%.15g", *(double*)data); break;
	}
}

static void RenderHexEditDialog() {
	if (!g_hexEditDlgOpen) return;

	ImVec2 ws = GUI.GetWindowSize();
	ImGui::SetNextWindowFocus();
	ImGui::SetNextWindowSize(ImVec2(400, 150), ImGuiCond_Always);
	ImGui::SetNextWindowPos(ImVec2(ws.x / 2 - 200, ws.y / 2 - 75), ImGuiCond_Appearing);
	ImGui::GetBackgroundDrawList()->AddRectFilled(ImVec2(0, 0), ws, IM_COL32(0, 0, 0, 100));

	char title[64];
	sprintf(title, "Edit %s at %s%llX", GetHexViewName(g_hexEditDlgType), g_showHexPrefix ? "0x" : "", g_hexEditDlgAddr);

	if (ImGui::Begin(title, &g_hexEditDlgOpen, ImGuiWindowFlags_NoResize | ImGuiWindowFlags_NoCollapse)) {
		ImGui::Text("Address: %s%llX", g_showHexPrefix ? "0x" : "", g_hexEditDlgAddr);
		ImGui::Text("Type: %s", GetHexViewName(g_hexEditDlgType));
		ImGui::Separator();

		ImGui::Text("Value:");
		ImGui::SetNextItemWidth(-1);
		bool apply = ImGui::InputText("##hev", g_hexEditDlgBuf, sizeof(g_hexEditDlgBuf),
			ImGuiInputTextFlags_EnterReturnsTrue | ImGuiInputTextFlags_AutoSelectAll);

		if (ImGui::IsWindowAppearing()) ImGui::SetKeyboardFocusHere(-1);

		ImGui::Separator();

		if (ImGui::Button("Apply", ImVec2(80, 0)) || apply) {
			auto& c = Core::I();
			int sz = GetHexViewSize(g_hexEditDlgType);
			u8 oldData[8] = { 0 }, newData[8] = { 0 };

			// Read old data
			for (int i = 0; i < sz && c.mem.InRange(g_hexEditDlgAddr + i); i++) {
				oldData[i] = c.mem.Read8(g_hexEditDlgAddr + i);
			}

			bool valid = true;
			switch (g_hexEditDlgType) {
			case HexViewType::ByteHex: {
				u32 v; if (sscanf(g_hexEditDlgBuf, "%x", &v) == 1) newData[0] = (u8)v; else valid = false;
				break;
			}
			case HexViewType::ByteDec: {
				int v; if (sscanf(g_hexEditDlgBuf, "%d", &v) == 1) newData[0] = (u8)v; else valid = false;
				break;
			}
			case HexViewType::Word2Hex: {
				u32 v; if (sscanf(g_hexEditDlgBuf, "%x", &v) == 1) *(u16*)newData = (u16)v; else valid = false;
				break;
			}
			case HexViewType::Word2Dec: {
				int v; if (sscanf(g_hexEditDlgBuf, "%d", &v) == 1) *(u16*)newData = (u16)v; else valid = false;
				break;
			}
			case HexViewType::Word4Hex: {
				u32 v; if (sscanf(g_hexEditDlgBuf, "%x", &v) == 1) *(u32*)newData = v; else valid = false;
				break;
			}
			case HexViewType::Word4Dec: {
				u32 v; if (sscanf(g_hexEditDlgBuf, "%u", &v) == 1) *(u32*)newData = v; else valid = false;
				break;
			}
			case HexViewType::Word8Hex: {
				u64 v; if (sscanf(g_hexEditDlgBuf, "%llx", &v) == 1) *(u64*)newData = v; else valid = false;
				break;
			}
			case HexViewType::Word8Dec: {
				u64 v; if (sscanf(g_hexEditDlgBuf, "%llu", &v) == 1) *(u64*)newData = v; else valid = false;
				break;
			}
			case HexViewType::Float: {
				float v; if (sscanf(g_hexEditDlgBuf, "%f", &v) == 1) *(float*)newData = v; else valid = false;
				break;
			}
			case HexViewType::Double: {
				double v; if (sscanf(g_hexEditDlgBuf, "%lf", &v) == 1) *(double*)newData = v; else valid = false;
				break;
			}
			}

			if (valid) {
				for (int i = 0; i < sz; i++) {
					if (c.mem.InRange(g_hexEditDlgAddr + i)) {
						c.mem.Write8(g_hexEditDlgAddr + i, newData[i]);
					}
				}
				c.undo.Record(g_hexEditDlgAddr, oldData, newData, sz);
				std::vector<u8> ov(oldData, oldData + sz), nv(newData, newData + sz);
				RecordEditHistory(g_hexEditDlgAddr, ov, nv);
				Log("Edited %s%llX: %s", g_showHexPrefix ? "0x" : "", g_hexEditDlgAddr, g_hexEditDlgBuf);
				g_hexEditDlgOpen = false;
			}
			else {
				Log("Invalid value: %s", g_hexEditDlgBuf);
			}
		}
		ImGui::SameLine();
		if (ImGui::Button("Cancel", ImVec2(80, 0)) || ImGui::IsKeyPressed(ImGuiKey_Escape)) {
			g_hexEditDlgOpen = false;
		}
	}
	ImGui::End();
}

// === HEXDUMP - Tamamen Yeniden Yazıldı ===
static void RenderHexDump() {
	auto& c = Core::I();
	ImVec2 reg = ImGui::GetContentRegionAvail();
	float sbw = 18, cw = reg.x - sbw - 4;
	bool dop = IsAnyDialogOpen() || g_hexEditDlgOpen;
	ImFont* ft = GetHexFont();
	float fs = GetHexFontSize();

	ImGui::PushAllowKeyboardFocus(false);
	ImGui::BeginChild("##hc", ImVec2(cw, 0), true, ImGuiWindowFlags_NoScrollbar | ImGuiWindowFlags_NoScrollWithMouse | ImGuiWindowFlags_NoNav);

	if (ImGui::IsWindowHovered() && ImGui::IsMouseClicked(0) && !dop)
		g_focusedPanel = FOCUS_HEXDUMP;
	bool isFoc = (g_focusedPanel == FOCUS_HEXDUMP) && !dop;

	int elemSize = GetHexViewSize(g_hexViewType);
	float charW = ft->CalcTextSizeA(fs, FLT_MAX, 0, "0").x;

	// Otomatik B/R hesaplama
	if (g_hexAutoRowBytes) {
		float addrW = charW * (c.loaded && c.pe.is64 ? 18 : 10);

		float elemW;
		switch (g_hexViewType) {
		case HexViewType::ByteHex: elemW = charW * 3; break;
		case HexViewType::ByteDec: elemW = charW * 4; break;
		case HexViewType::Word2Hex: elemW = charW * 5.5f; break;
		case HexViewType::Word2Dec: elemW = charW * 6.5f; break;
		case HexViewType::Word4Hex: elemW = charW * 9.5f; break;
		case HexViewType::Word4Dec: elemW = charW * 11; break;
		case HexViewType::Word8Hex: elemW = charW * 17.5f; break;
		case HexViewType::Word8Dec: elemW = charW * 21; break;
		case HexViewType::Float: elemW = charW * 12; break;
		case HexViewType::Double: elemW = charW * 12; break;
		default: elemW = charW * 3; break;
		}

		float perByteWidth = (elemW / elemSize) + charW;
		float availW = cw - addrW - charW * 6;
		int calcBPR = (int)(availW / perByteWidth);
		calcBPR = (calcBPR / elemSize) * elemSize;
		calcBPR = std::clamp(calcBPR, elemSize, 64);
		g_hexBytesPerRow = calcBPR;
	}

	int effectiveBPR = g_hexBytesPerRow;
	if (g_hexViewType == HexViewType::Float || g_hexViewType == HexViewType::Double) {
		effectiveBPR = (g_hexBytesPerRow / elemSize) * elemSize;
		if (effectiveBPR < elemSize) effectiveBPR = elemSize;
	}

	// Header
	ImGui::TextColored(ImVec4(0.9f, 0.6f, 0.3f, 1), "HexDump - %s%llX [%s]",
		g_showHexPrefix ? "0x" : "", g_hexAddr, GetHexViewName(g_hexViewType));
	ImGui::SameLine();

	ImGui::Checkbox("Auto", &g_hexAutoRowBytes);
	if (ImGui::IsItemHovered()) ImGui::SetTooltip("Automatically calculate bytes per row");
	ImGui::SameLine();

	if (g_hexAutoRowBytes) ImGui::BeginDisabled();
	ImGui::Text("B/R:"); ImGui::SameLine(); ImGui::SetNextItemWidth(40);
	if (ImGui::InputInt("##bp", &g_hexBytesPerRow, 0, 0))
		g_hexBytesPerRow = std::clamp(g_hexBytesPerRow, 4, 64);
	if (g_hexAutoRowBytes) ImGui::EndDisabled();

	ImGui::SameLine();
	const char* sm[] = { "Step","Line" };
	ImGui::SetNextItemWidth(50);
	ImGui::Combo("##sm", &g_hexScrollMode, sm, 2);
	ImGui::Separator();

	// Context menu
	if (ImGui::BeginPopupContextWindow("HexContextMenu")) {
		ImGui::TextDisabled("View Type:");
		ImGui::Separator();
		if (ImGui::MenuItem("Byte Hex", nullptr, g_hexViewType == HexViewType::ByteHex)) g_hexViewType = HexViewType::ByteHex;
		if (ImGui::MenuItem("Byte Decimal", nullptr, g_hexViewType == HexViewType::ByteDec)) g_hexViewType = HexViewType::ByteDec;
		ImGui::Separator();
		if (ImGui::MenuItem("2 Byte Hex", nullptr, g_hexViewType == HexViewType::Word2Hex)) g_hexViewType = HexViewType::Word2Hex;
		if (ImGui::MenuItem("2 Byte Decimal", nullptr, g_hexViewType == HexViewType::Word2Dec)) g_hexViewType = HexViewType::Word2Dec;
		ImGui::Separator();
		if (ImGui::MenuItem("4 Byte Hex", nullptr, g_hexViewType == HexViewType::Word4Hex)) g_hexViewType = HexViewType::Word4Hex;
		if (ImGui::MenuItem("4 Byte Decimal", nullptr, g_hexViewType == HexViewType::Word4Dec)) g_hexViewType = HexViewType::Word4Dec;
		ImGui::Separator();
		if (ImGui::MenuItem("8 Byte Hex", nullptr, g_hexViewType == HexViewType::Word8Hex)) g_hexViewType = HexViewType::Word8Hex;
		if (ImGui::MenuItem("8 Byte Decimal", nullptr, g_hexViewType == HexViewType::Word8Dec)) g_hexViewType = HexViewType::Word8Dec;
		ImGui::Separator();
		if (ImGui::MenuItem("Float", nullptr, g_hexViewType == HexViewType::Float)) g_hexViewType = HexViewType::Float;
		if (ImGui::MenuItem("Double", nullptr, g_hexViewType == HexViewType::Double)) g_hexViewType = HexViewType::Double;
		ImGui::EndPopup();
	}

	float lh = fs + 4, ah = ImGui::GetContentRegionAvail().y;
	int vr = std::max((int)(ah / lh), 5);
	int bpr = effectiveBPR;
	int elemsPerRow = bpr / elemSize;

	ImGuiIO& io = ImGui::GetIO();
	ImDrawList* dl = ImGui::GetWindowDrawList();

	if (ImGui::IsMouseReleased(0)) g_hexDragging = false;
	ImGui::PushStyleVar(ImGuiStyleVar_ItemSpacing, ImVec2(0, 0));

	u64 addr = g_hexAddr;

	// Element width
	float elemW;
	switch (g_hexViewType) {
	case HexViewType::ByteHex: elemW = charW * 3; break;
	case HexViewType::ByteDec: elemW = charW * 4; break;
	case HexViewType::Word2Hex: elemW = charW * 5.5f; break;
	case HexViewType::Word2Dec: elemW = charW * 6.5f; break;
	case HexViewType::Word4Hex: elemW = charW * 9.5f; break;
	case HexViewType::Word4Dec: elemW = charW * 11; break;
	case HexViewType::Word8Hex: elemW = charW * 17.5f; break;
	case HexViewType::Word8Dec: elemW = charW * 21; break;
	case HexViewType::Float: elemW = charW * 12; break;
	case HexViewType::Double: elemW = charW * 12; break;
	default: elemW = charW * 3; break;
	}

	float elemH = lh - 2;
	ImU32 selActive = (g_themeMode == ThemeMode::Dark) ? IM_COL32(60, 80, 150, 255) : IM_COL32(100, 140, 200, 255);
	ImU32 selDim = (g_themeMode == ThemeMode::Dark) ? IM_COL32(80, 60, 100, 255) : IM_COL32(140, 120, 180, 180);
	ImU32 nibbleIndicatorColor = IM_COL32(255, 200, 50, 255);
	ImU32 asciiTextColor = (g_themeMode == ThemeMode::Dark) ? IM_COL32(255, 255, 255, 255) : IM_COL32(0, 0, 0, 255);
	ImU32 asciiCursorColor = IM_COL32(255, 100, 100, 255);

	u64 selStart, selEnd;
	GetHexSelection(selStart, selEnd);

	for (int row = 0; row < vr; row++) {
		ImVec2 rs = ImGui::GetCursorScreenPos();
		std::string as = FormatDisplayAddress(addr, c.loaded ? c.pe.is64 : true);
		dl->AddText(ft, fs, rs, g_colors.White, as.c_str());
		float addrW = ft->CalcTextSizeA(fs, FLT_MAX, 0, as.c_str()).x;
		float dataX = rs.x + addrW + charW * 2;

		u8 rowData[64] = { 0 };
		bool rowValid[64] = { false };
		for (int i = 0; i < bpr; i++) {
			if (c.loaded && c.mem.InRange(addr + i)) {
				rowData[i] = c.mem.Read8(addr + i);
				rowValid[i] = true;
			}
		}

		// Data elements
		for (int e = 0; e < elemsPerRow; e++) {
			int byteOff = e * elemSize;
			u64 elemAddr = addr + byteOff;
			float xPos = dataX + e * elemW;
			ImVec2 ep(xPos, rs.y);

			bool isSelected = g_hexHasSelection && elemAddr >= selStart && elemAddr <= selEnd;
			bool isCursorHere = (elemAddr == g_hexCursor);

			if (isSelected) {
				ImU32 highlightColor = (!g_hexSelectionFromAscii) ? selActive : selDim;
				dl->AddRectFilled(ep, ImVec2(ep.x + elemW - 2, ep.y + elemH), highlightColor);
			}

			bool valid = true;
			for (int b = 0; b < elemSize; b++) {
				if (!rowValid[byteOff + b]) { valid = false; break; }
			}

			char valStr[32] = "??";
			if (valid) {
				switch (g_hexViewType) {
				case HexViewType::ByteHex: sprintf(valStr, "%02X", rowData[byteOff]); break;
				case HexViewType::ByteDec: sprintf(valStr, "%3d", rowData[byteOff]); break;
				case HexViewType::Word2Hex: { u16 v = *(u16*)&rowData[byteOff]; sprintf(valStr, "%04X", v); break; }
				case HexViewType::Word2Dec: { u16 v = *(u16*)&rowData[byteOff]; sprintf(valStr, "%5d", v); break; }
				case HexViewType::Word4Hex: { u32 v = *(u32*)&rowData[byteOff]; sprintf(valStr, "%08X", v); break; }
				case HexViewType::Word4Dec: { u32 v = *(u32*)&rowData[byteOff]; sprintf(valStr, "%10u", v); break; }
				case HexViewType::Word8Hex: { u64 v = *(u64*)&rowData[byteOff]; sprintf(valStr, "%016llX", v); break; }
				case HexViewType::Word8Dec: { u64 v = *(u64*)&rowData[byteOff]; sprintf(valStr, "%20llu", v); break; }
				case HexViewType::Float: {
					float v = *(float*)&rowData[byteOff];
					if (std::isnan(v) || std::isinf(v)) strcpy(valStr, "NaN");
					else { sprintf(valStr, "%.6g", v); if (strlen(valStr) > 8) sprintf(valStr, "%.2e", v); }
					break;
				}
				case HexViewType::Double: {
					double v = *(double*)&rowData[byteOff];
					if (std::isnan(v) || std::isinf(v)) strcpy(valStr, "NaN");
					else { sprintf(valStr, "%.6g", v); if (strlen(valStr) > 10) sprintf(valStr, "%.2e", v); }
					break;
				}
				}
			}
			dl->AddText(ft, fs, ImVec2(ep.x + 2, ep.y), valid ? g_colors.White : g_colors.Gray, valStr);

			if (isCursorHere && g_hexHasSelection && !g_hexSelectionFromAscii && g_hexViewType == HexViewType::ByteHex) {
				float indicatorX = g_hexHighNibble ? (ep.x + 1) : (ep.x + charW + 1);
				dl->AddLine(
					ImVec2(indicatorX, ep.y + 1),
					ImVec2(indicatorX, ep.y + elemH - 1),
					nibbleIndicatorColor, 2.0f
				);
			}

			ImVec2 elemMin = ep, elemMax(ep.x + elemW - 2, ep.y + elemH);
			if (ImGui::IsMouseHoveringRect(elemMin, elemMax) && !dop) {
				if (ImGui::IsMouseClicked(0)) {
					g_hexSelectionFromAscii = false;
					if (io.KeyShift && g_hexHasSelection) {
						g_hexCursor = elemAddr;
					}
					else {
						g_hexAnchor = g_hexCursor = elemAddr;
						g_hexHasSelection = true;
						g_hexHighNibble = true;
					}
					g_hexDragging = true;
					g_focusedPanel = FOCUS_HEXDUMP;
				}
				if (ImGui::IsMouseDoubleClicked(0)) {
					OpenHexEditDialog(elemAddr, g_hexViewType);
				}
				if (g_hexDragging && ImGui::IsMouseDown(0)) {
					g_hexCursor = elemAddr;
				}
			}
		}

		// ASCII section
		float asciiX = dataX + elemsPerRow * elemW + charW * 2;
		dl->AddText(ft, fs, ImVec2(asciiX, rs.y), g_colors.White, "|");
		asciiX += charW * 1.5f;

		for (int i = 0; i < bpr; i++) {
			u64 ba = addr + i;
			bool isSel = g_hexHasSelection && ba >= selStart && ba <= selEnd;
			bool isCursorHere = (ba == g_hexCursor);
			char ch = '.';
			if (rowValid[i] && rowData[i] >= 32 && rowData[i] < 127) ch = (char)rowData[i];
			ImVec2 cp(asciiX + i * charW, rs.y);

			if (isSel) {
				ImU32 highlightColor = g_hexSelectionFromAscii ? selActive : selDim;
				dl->AddRectFilled(cp, ImVec2(cp.x + charW, cp.y + elemH), highlightColor);
			}

			char cs[2] = { ch, 0 };
			dl->AddText(ft, fs, cp, rowValid[i] ? asciiTextColor : g_colors.Gray, cs);

			if (isCursorHere && g_hexHasSelection && g_hexSelectionFromAscii) {
				dl->AddLine(
					ImVec2(cp.x, cp.y + elemH - 1),
					ImVec2(cp.x + charW, cp.y + elemH - 1),
					asciiCursorColor, 2.0f
				);
			}

			ImVec2 asciiMin = cp, asciiMax(cp.x + charW, cp.y + elemH);
			if (ImGui::IsMouseHoveringRect(asciiMin, asciiMax) && !dop) {
				if (ImGui::IsMouseClicked(0)) {
					g_hexSelectionFromAscii = true;
					if (io.KeyShift && g_hexHasSelection) {
						g_hexCursor = ba;
					}
					else {
						g_hexAnchor = g_hexCursor = ba;
						g_hexHasSelection = true;
					}
					g_hexDragging = true;
					g_focusedPanel = FOCUS_HEXDUMP;
				}
				if (g_hexDragging && ImGui::IsMouseDown(0)) {
					g_hexCursor = ba;
				}
			}
		}
		dl->AddText(ft, fs, ImVec2(asciiX + bpr * charW, rs.y), g_colors.White, "|");

		ImGui::SetCursorScreenPos(ImVec2(rs.x, rs.y + lh));
		addr += bpr;
	}
	ImGui::PopStyleVar();

	// Keyboard handling
	if (isFoc && g_hexHasSelection) {
		bool shift = io.KeyShift;

		// DÜZELTME: Önce cursor'ın görünür alanda olup olmadığını kontrol et
		// Eğer değilse, cursor'ı görünür alana getir
		u64 viewEnd = g_hexAddr + (u64)(vr * bpr);
		bool cursorVisible = (g_hexCursor >= g_hexAddr && g_hexCursor < viewEnd);

		// Herhangi bir navigation tuşuna basıldığında cursor görünür değilse önce görünür yap
		bool anyNavKey = IsKeyPressedManual(ImGuiKey_UpArrow) || IsKeyPressedManual(ImGuiKey_DownArrow) ||
			IsKeyPressedManual(ImGuiKey_LeftArrow) || IsKeyPressedManual(ImGuiKey_RightArrow);

		if (anyNavKey && !cursorVisible) {
			// Cursor'ı görünür alana getir
			if (g_hexCursor < g_hexAddr) {
				// Cursor yukarıda
				g_hexAddr = (g_hexCursor / bpr) * bpr;
			}
			else {
				// Cursor aşağıda
				g_hexAddr = ((g_hexCursor / bpr) - vr + 2) * bpr;
				if ((i64)g_hexAddr < 0) g_hexAddr = 0;
			}
			// Tuş işleme normal devam edecek
		}

		if (ImGui::IsKeyPressed(ImGuiKey_Enter, false)) {
			OpenHexEditDialog(g_hexCursor, g_hexViewType);
		}

		// Navigation
		if (g_hexSelectionFromAscii) {
			if (IsKeyPressedManual(ImGuiKey_LeftArrow)) {
				if (g_hexCursor > 0) g_hexCursor -= 1;
				if (!shift) g_hexAnchor = g_hexCursor;
				if (g_hexCursor < g_hexAddr) g_hexAddr = (g_hexCursor / bpr) * bpr;
			}
			if (IsKeyPressedManual(ImGuiKey_RightArrow)) {
				g_hexCursor += 1;
				if (!shift) g_hexAnchor = g_hexCursor;
				if (g_hexCursor >= g_hexAddr + (u64)(vr * bpr)) g_hexAddr += bpr;
			}
		}
		else if (g_hexViewType == HexViewType::ByteHex) {
			if (IsKeyPressedManual(ImGuiKey_RightArrow)) {
				if (g_hexHighNibble) {
					g_hexHighNibble = false;
				}
				else {
					g_hexCursor += 1;
					g_hexHighNibble = true;
				}
				if (!shift) g_hexAnchor = g_hexCursor;
				if (g_hexCursor >= g_hexAddr + (u64)(vr * bpr)) g_hexAddr += bpr;
			}
			if (IsKeyPressedManual(ImGuiKey_LeftArrow)) {
				if (!g_hexHighNibble) {
					g_hexHighNibble = true;
				}
				else {
					if (g_hexCursor > 0) {
						g_hexCursor -= 1;
						g_hexHighNibble = false;
					}
				}
				if (!shift) g_hexAnchor = g_hexCursor;
				if (g_hexCursor < g_hexAddr) g_hexAddr = (g_hexCursor / bpr) * bpr;
			}
		}
		else {
			if (IsKeyPressedManual(ImGuiKey_LeftArrow)) {
				if (g_hexCursor >= (u64)elemSize) {
					g_hexCursor -= elemSize;
					if (!shift) g_hexAnchor = g_hexCursor;
				}
				if (g_hexCursor < g_hexAddr) g_hexAddr = (g_hexCursor / bpr) * bpr;
			}
			if (IsKeyPressedManual(ImGuiKey_RightArrow)) {
				g_hexCursor += elemSize;
				if (!shift) g_hexAnchor = g_hexCursor;
				if (g_hexCursor >= g_hexAddr + (u64)(vr * bpr)) g_hexAddr += bpr;
			}
		}

		// Yukarı/aşağı
		if (IsKeyPressedManual(ImGuiKey_UpArrow)) {
			if (g_hexCursor >= (u64)bpr) {
				g_hexCursor -= bpr;
				if (!shift) g_hexAnchor = g_hexCursor;
			}
			if (g_hexCursor < g_hexAddr)
				g_hexAddr = (g_hexCursor / bpr) * bpr;
		}
		if (IsKeyPressedManual(ImGuiKey_DownArrow)) {
			g_hexCursor += bpr;
			if (!shift) g_hexAnchor = g_hexCursor;
			if (g_hexCursor >= g_hexAddr + (u64)(vr * bpr)) g_hexAddr += bpr;
		}

		// Direct keyboard editing
		if (!io.KeyCtrl && c.loaded && c.mem.InRange(g_hexCursor)) {
			if (g_hexSelectionFromAscii) {
				for (int ch = 32; ch < 127; ch++) {
					ImGuiKey key = ImGuiKey_None;

					if (ch >= 'a' && ch <= 'z') key = (ImGuiKey)(ImGuiKey_A + (ch - 'a'));
					else if (ch >= 'A' && ch <= 'Z') key = (ImGuiKey)(ImGuiKey_A + (ch - 'A'));
					else if (ch >= '0' && ch <= '9') key = (ImGuiKey)(ImGuiKey_0 + (ch - '0'));
					else if (ch == ' ') key = ImGuiKey_Space;

					if (key != ImGuiKey_None && IsKeyPressedManual(key)) {
						u8 newChar = (u8)ch;

						if (io.KeyShift && ch >= 'a' && ch <= 'z') {
							newChar = (u8)(ch - 32);
						}
						else if (!io.KeyShift && ch >= 'A' && ch <= 'Z') {
							newChar = (u8)(ch + 32);
						}

						u8 old = c.mem.Read8(g_hexCursor);
						c.mem.Write8(g_hexCursor, newChar);
						std::vector<u8> ov = { old }, nv = { newChar };
						c.undo.Record(g_hexCursor, ov.data(), nv.data(), 1);
						RecordEditHistory(g_hexCursor, ov, nv);

						g_hexCursor += 1;
						g_hexAnchor = g_hexCursor;
						if (g_hexCursor >= g_hexAddr + (u64)(vr * bpr)) g_hexAddr += bpr;
						break;
					}
				}
			}
			else if (g_hexViewType == HexViewType::ByteHex) {
				for (int k = 0; k <= 15; k++) {
					ImGuiKey key = (k < 10) ? (ImGuiKey)(ImGuiKey_0 + k) : (ImGuiKey)(ImGuiKey_A + (k - 10));
					if (IsKeyPressedManual(key)) {
						u8 old = c.mem.Read8(g_hexCursor);
						u8 nv;

						if (g_hexHighNibble) {
							nv = (k << 4) | (old & 0x0F);
							c.mem.Write8(g_hexCursor, nv);
							std::vector<u8> ov = { old }, nvv = { nv };
							c.undo.Record(g_hexCursor, ov.data(), nvv.data(), 1);
							RecordEditHistory(g_hexCursor, ov, nvv);
							g_hexHighNibble = false;
						}
						else {
							nv = (old & 0xF0) | k;
							c.mem.Write8(g_hexCursor, nv);
							std::vector<u8> ov = { old }, nvv = { nv };
							c.undo.Record(g_hexCursor, ov.data(), nvv.data(), 1);
							RecordEditHistory(g_hexCursor, ov, nvv);
							g_hexCursor += 1;
							g_hexAnchor = g_hexCursor;
							g_hexHighNibble = true;
							if (g_hexCursor >= g_hexAddr + (u64)(vr * bpr)) g_hexAddr += bpr;
						}
						break;
					}
				}
			}
		}

		// Copy/Paste (mevcut kod aynı)
		if (io.KeyCtrl && ImGui::IsKeyPressed(ImGuiKey_C, false) && c.loaded) {
			u64 ss, se;
			GetHexSelection(ss, se);
			std::string copyStr;

			if (g_hexSelectionFromAscii) {
				for (u64 a = ss; a <= se; a++) {
					if (c.mem.InRange(a)) {
						u8 b = c.mem.Read8(a);
						if (b >= 32 && b < 127) copyStr += (char)b;
						else copyStr += '.';
					}
				}
				Log("Copied ASCII: %s", copyStr.c_str());
			}
			else {
				u64 a = ss;
				while (a <= se) {
					if (!c.mem.InRange(a)) { a += elemSize; continue; }

					u8 data[8] = { 0 };
					bool valid = true;
					for (int i = 0; i < elemSize && a + i <= se; i++) {
						if (c.mem.InRange(a + i)) data[i] = c.mem.Read8(a + i);
						else valid = false;
					}

					if (valid) {
						char buf[32];
						switch (g_hexViewType) {
						case HexViewType::ByteHex: sprintf(buf, "%02X ", data[0]); break;
						case HexViewType::ByteDec: sprintf(buf, "%d ", data[0]); break;
						case HexViewType::Word2Hex: sprintf(buf, "%04X ", *(u16*)data); break;
						case HexViewType::Word2Dec: sprintf(buf, "%d ", *(u16*)data); break;
						case HexViewType::Word4Hex: sprintf(buf, "%08X ", *(u32*)data); break;
						case HexViewType::Word4Dec: sprintf(buf, "%u ", *(u32*)data); break;
						case HexViewType::Word8Hex: sprintf(buf, "%016llX ", *(u64*)data); break;
						case HexViewType::Word8Dec: sprintf(buf, "%llu ", *(u64*)data); break;
						case HexViewType::Float: sprintf(buf, "%.6g ", *(float*)data); break;
						case HexViewType::Double: sprintf(buf, "%.15g ", *(double*)data); break;
						}
						copyStr += buf;
					}
					a += elemSize;
				}
				if (!copyStr.empty() && copyStr.back() == ' ') copyStr.pop_back();
				Log("Copied Hex: %s", copyStr.c_str());
			}

			if (!copyStr.empty()) {
				ImGui::SetClipboardText(copyStr.c_str());
			}
		}

		// Paste
		if (io.KeyCtrl && ImGui::IsKeyPressed(ImGuiKey_V, false) && c.loaded) {
			const char* clip = ImGui::GetClipboardText();
			if (clip) {
				std::string s = clip;
				u64 wa = g_hexCursor;
				std::vector<u8> oldBytes, newBytes;

				if (g_hexSelectionFromAscii) {
					// ASCII paste - karakter olarak
					for (char ch : s) {
						if (c.mem.InRange(wa)) {
							oldBytes.push_back(c.mem.Read8(wa));
							newBytes.push_back((u8)ch);
							c.mem.Write8(wa, (u8)ch);
							wa++;
						}
					}
				}
				else {
					// Hex paste
					std::istringstream iss(s);
					std::string token;
					while (iss >> token) {
						u8 data[8] = { 0 };
						bool valid = false;

						switch (g_hexViewType) {
						case HexViewType::ByteHex: {
							u32 v; if (sscanf(token.c_str(), "%x", &v) == 1) { data[0] = (u8)v; valid = true; }
							break;
						}
						case HexViewType::ByteDec: {
							int v; if (sscanf(token.c_str(), "%d", &v) == 1) { data[0] = (u8)v; valid = true; }
							break;
						}
						case HexViewType::Word2Hex: {
							u32 v; if (sscanf(token.c_str(), "%x", &v) == 1) { *(u16*)data = (u16)v; valid = true; }
							break;
						}
						case HexViewType::Word2Dec: {
							int v; if (sscanf(token.c_str(), "%d", &v) == 1) { *(u16*)data = (u16)v; valid = true; }
							break;
						}
						case HexViewType::Word4Hex: {
							u32 v; if (sscanf(token.c_str(), "%x", &v) == 1) { *(u32*)data = v; valid = true; }
							break;
						}
						case HexViewType::Word4Dec: {
							u32 v; if (sscanf(token.c_str(), "%u", &v) == 1) { *(u32*)data = v; valid = true; }
							break;
						}
						case HexViewType::Word8Hex: {
							u64 v; if (sscanf(token.c_str(), "%llx", &v) == 1) { *(u64*)data = v; valid = true; }
							break;
						}
						case HexViewType::Word8Dec: {
							u64 v; if (sscanf(token.c_str(), "%llu", &v) == 1) { *(u64*)data = v; valid = true; }
							break;
						}
						case HexViewType::Float: {
							float v; if (sscanf(token.c_str(), "%f", &v) == 1) { *(float*)data = v; valid = true; }
							break;
						}
						case HexViewType::Double: {
							double v; if (sscanf(token.c_str(), "%lf", &v) == 1) { *(double*)data = v; valid = true; }
							break;
						}
						}

						if (valid) {
							for (int i = 0; i < elemSize; i++) {
								if (c.mem.InRange(wa + i)) {
									oldBytes.push_back(c.mem.Read8(wa + i));
									newBytes.push_back(data[i]);
									c.mem.Write8(wa + i, data[i]);
								}
							}
							wa += elemSize;
						}
					}
				}

				if (!oldBytes.empty()) {
					c.undo.Record(g_hexCursor, oldBytes.data(), newBytes.data(), oldBytes.size());
					RecordEditHistory(g_hexCursor, oldBytes, newBytes);
					Log("Pasted %zu bytes", newBytes.size());
				}
			}
		}
	}

	// Mouse wheel
	if (ImGui::IsWindowHovered() && !dop) {
		float w = io.MouseWheel;
		if (w != 0) {
			i64 d = (g_hexScrollMode == 0) ? (i64)(-w * 4) : (i64)(-w) * bpr;
			g_hexAddr = (u64)std::max((i64)0, (i64)g_hexAddr + d);
		}
	}

	ImGui::EndChild();
	ImGui::PopAllowKeyboardFocus();

	// Scrollbar
	ImGui::SameLine();
	ImVec2 sp = ImGui::GetCursorScreenPos();
	int sd = 0;
	RenderCustomScrollbar("##hs", sp.x, sp.y, sbw, reg.y, &sd);
	if (sd != 0 && !dop) {
		g_hexAddr = (u64)std::max((i64)0, (i64)g_hexAddr + (g_hexScrollMode == 0 ? sd : sd * bpr));
	}
	ImGui::Dummy(ImVec2(sbw, reg.y));
}

static void RenderFreeMemDlg() {
	if (!g_freeMemDlgOpen) return;

	auto& c = Core::I();
	ImVec2 ws = GUI.GetWindowSize();

	ImGui::SetNextWindowSize(ImVec2(450, 300), ImGuiCond_FirstUseEver);
	ImGui::SetNextWindowPos(ImVec2(ws.x / 2 - 225, ws.y / 2 - 150), ImGuiCond_Appearing);

	if (ImGui::Begin("Allocate Free Memory", &g_freeMemDlgOpen, ImGuiWindowFlags_NoCollapse)) {
		if (!c.loaded) {
			ImGui::TextColored(ImVec4(1, 0.5f, 0.5f, 1), "No file loaded!");
			ImGui::End();
			return;
		}

		ImGui::TextColored(ImVec4(0.9f, 0.6f, 0.3f, 1), "Allocation Mode");
		ImGui::Separator();

		ImGui::RadioButton("New Section", &g_freeMemMode, 0);
		if (ImGui::IsItemHovered()) {
			ImGui::SetTooltip("Create a new PE section with the specified size.\nThis is the standard and safest method.");
		}

		ImGui::SameLine();
		ImGui::RadioButton("Code Cave", &g_freeMemMode, 1);
		if (ImGui::IsItemHovered()) {
			ImGui::SetTooltip("Find and use existing empty space (INT3/NOP padding).\nNo new section, but limited size.");
		}

		ImGui::Spacing();
		ImGui::TextColored(ImVec4(0.9f, 0.6f, 0.3f, 1), "Parameters");
		ImGui::Separator();

		ImGui::Text("Size (bytes):");
		ImGui::SameLine();
		ImGui::SetNextItemWidth(150);
		ImGui::InputText("##fms", g_freeMemSizeBuf, sizeof(g_freeMemSizeBuf));
		ImGui::SameLine();
		if (ImGui::SmallButton("256")) strcpy(g_freeMemSizeBuf, "256");
		ImGui::SameLine();
		if (ImGui::SmallButton("1K")) strcpy(g_freeMemSizeBuf, "1024");
		ImGui::SameLine();
		if (ImGui::SmallButton("4K")) strcpy(g_freeMemSizeBuf, "4096");

		if (g_freeMemMode == 0) {
			ImGui::Text("Section Name:");
			ImGui::SameLine();
			ImGui::SetNextItemWidth(150);
			ImGui::InputText("##fmn", g_freeMemNameBuf, sizeof(g_freeMemNameBuf));

			// Section name validation
			size_t nameLen = strlen(g_freeMemNameBuf);
			if (nameLen > 8) {
				ImGui::TextColored(ImVec4(1, 0.5f, 0.5f, 1), "Name too long (max 8 chars)");
			}
		}

		ImGui::Spacing();
		ImGui::Separator();

		// Preview
		size_t reqSize = 0;
		try {
			std::string sizeStr = g_freeMemSizeBuf;
			if (sizeStr.find("0x") == 0 || sizeStr.find("0X") == 0) {
				reqSize = (size_t)strtoull(sizeStr.c_str(), nullptr, 16);
			}
			else {
				reqSize = (size_t)strtoull(sizeStr.c_str(), nullptr, 0);
			}
		}
		catch (...) {}

		if (g_freeMemMode == 0) {
			// Section mode preview
			u64 newSectionVA = 0;
			if (!c.pe.secs.empty()) {
				auto& lastSec = c.pe.secs.back();
				u64 lastEnd = lastSec.va + lastSec.vSz;
				u64 alignment = c.pe.is64 ? 0x1000 : 0x1000;
				newSectionVA = ((lastEnd + alignment - 1) / alignment) * alignment;
			}

			ImGui::Text("New section will be created at: %s%llX",
				g_showHexPrefix ? "0x" : "", c.Base() + newSectionVA);
			ImGui::Text("Size: %zu bytes (0x%zX)", reqSize, reqSize);
		}
		else {
			// Code cave mode preview - INT3 bloklarını tara
			ImGui::Text("Scanning for code caves...");

			size_t largestCave = 0;
			u64 largestCaveAddr = 0;

			// .text section'da INT3 bloklarını bul
			for (auto& sec : c.pe.secs) {
				if (!(sec.chars & 0x20000000)) continue;  // Executable değilse atla

				u64 secStart = c.Base() + sec.va;
				u64 secEnd = secStart + sec.vSz;

				size_t currentCave = 0;
				u64 currentCaveStart = 0;

				for (u64 addr = secStart; addr < secEnd; addr++) {
					if (c.mem.InRange(addr)) {
						u8 b = c.mem.Read8(addr);
						if (b == 0xCC || b == 0x90) {  // INT3 or NOP
							if (currentCave == 0) currentCaveStart = addr;
							currentCave++;
						}
						else {
							if (currentCave > largestCave) {
								largestCave = currentCave;
								largestCaveAddr = currentCaveStart;
							}
							currentCave = 0;
						}
					}
				}
				if (currentCave > largestCave) {
					largestCave = currentCave;
					largestCaveAddr = currentCaveStart;
				}
			}

			if (largestCave > 0) {
				ImGui::TextColored(ImVec4(0.5f, 1, 0.5f, 1),
					"Largest cave: %zu bytes at %s%llX",
					largestCave, g_showHexPrefix ? "0x" : "", largestCaveAddr);

				if (reqSize > largestCave) {
					ImGui::TextColored(ImVec4(1, 0.5f, 0.5f, 1),
						"Requested size (%zu) exceeds largest cave!", reqSize);
				}
			}
			else {
				ImGui::TextColored(ImVec4(1, 0.5f, 0.5f, 1), "No suitable code caves found!");
			}
		}

		ImGui::Spacing();
		ImGui::Separator();

		// Buttons
		bool canAllocate = reqSize > 0;
		if (g_freeMemMode == 0) {
			canAllocate = canAllocate && strlen(g_freeMemNameBuf) > 0 && strlen(g_freeMemNameBuf) <= 8;
		}

		if (!canAllocate) ImGui::BeginDisabled();

		if (ImGui::Button("Allocate", ImVec2(100, 0))) {
			if (g_freeMemMode == 0) {
				// Section mode
				u64 newAddr = c.pe.AddSection(g_freeMemNameBuf, reqSize,
					0xE0000020);  // RWX + Code

				if (newAddr) {
					// Memory'yi genişlet
					c.mem.Expand(reqSize);

					// INT3 ile doldur
					for (size_t i = 0; i < reqSize; i++) {
						if (c.mem.InRange(newAddr + i)) {
							c.mem.Write8(newAddr + i, 0xCC);
						}
					}

					g_lastAllocatedAddr = newAddr;
					g_lastAllocatedSize = reqSize;

					Log("Allocated section '%s' at %s%llX - %s%llX (%zu bytes)",
						g_freeMemNameBuf,
						g_showHexPrefix ? "0x" : "", newAddr,
						g_showHexPrefix ? "0x" : "", newAddr + reqSize - 1,
						reqSize);

					// Go to new section
					g_viewAddr = newAddr;
					g_selAddr = newAddr;
					g_disPreBufferValid = false;

					g_freeMemDlgOpen = false;
				}
				else {
					Log("Failed to allocate section!");
				}
			}
			else {
				// Code cave mode - en büyük cave'i bul ve kullan
				size_t largestCave = 0;
				u64 largestCaveAddr = 0;

				for (auto& sec : c.pe.secs) {
					if (!(sec.chars & 0x20000000)) continue;

					u64 secStart = c.Base() + sec.va;
					u64 secEnd = secStart + sec.vSz;

					size_t currentCave = 0;
					u64 currentCaveStart = 0;

					for (u64 addr = secStart; addr < secEnd; addr++) {
						if (c.mem.InRange(addr)) {
							u8 b = c.mem.Read8(addr);
							if (b == 0xCC || b == 0x90) {
								if (currentCave == 0) currentCaveStart = addr;
								currentCave++;
							}
							else {
								if (currentCave > largestCave) {
									largestCave = currentCave;
									largestCaveAddr = currentCaveStart;
								}
								currentCave = 0;
							}
						}
					}
					if (currentCave > largestCave) {
						largestCave = currentCave;
						largestCaveAddr = currentCaveStart;
					}
				}

				if (largestCave >= reqSize) {
					g_lastAllocatedAddr = largestCaveAddr;
					g_lastAllocatedSize = reqSize;

					Log("Found code cave at %s%llX - %s%llX (%zu bytes available, using %zu)",
						g_showHexPrefix ? "0x" : "", largestCaveAddr,
						g_showHexPrefix ? "0x" : "", largestCaveAddr + largestCave - 1,
						largestCave, reqSize);

					g_viewAddr = largestCaveAddr;
					g_selAddr = largestCaveAddr;
					g_disPreBufferValid = false;

					g_freeMemDlgOpen = false;
				}
				else {
					Log("No code cave large enough! (need %zu, largest is %zu)", reqSize, largestCave);
				}
			}
		}

		if (!canAllocate) ImGui::EndDisabled();

		ImGui::SameLine();
		if (ImGui::Button("Cancel", ImVec2(100, 0))) {
			g_freeMemDlgOpen = false;
		}

		// Last allocation info
		if (g_lastAllocatedAddr != 0) {
			ImGui::Spacing();
			ImGui::Separator();
			ImGui::TextColored(ImVec4(0.5f, 1, 0.5f, 1), "Last allocation:");
			ImGui::Text("  Base: %s%llX", g_showHexPrefix ? "0x" : "", g_lastAllocatedAddr);
			ImGui::Text("  End:  %s%llX", g_showHexPrefix ? "0x" : "", g_lastAllocatedAddr + g_lastAllocatedSize - 1);
			ImGui::Text("  Size: %zu bytes", g_lastAllocatedSize);

			if (ImGui::SmallButton("Go to Base")) {
				g_viewAddr = g_lastAllocatedAddr;
				g_selAddr = g_lastAllocatedAddr;
				g_disPreBufferValid = false;
			}
			ImGui::SameLine();
			if (ImGui::SmallButton("Copy Base")) {
				char buf[32];
				sprintf(buf, "%s%llX", g_showHexPrefix ? "0x" : "", g_lastAllocatedAddr);
				ImGui::SetClipboardText(buf);
			}
		}
	}
	ImGui::End();
}

"0x" : "", f.iatAddr, f.name.c_str()); if (ImGui::Selectable(b) && !dop) { char l[128]; sprintf(l, "%s.%s", d.name.c_str(), f.name.c_str()); NavGotoWithSwitch(f.iatAddr, l, true, false); g_focusedPanel = FOCUS_DISASM; } ImGui::PopID(); } ImGui::TreePop(); }
		ImGui::PopID();
	}
}

// === ACTIONS ===
static void DoOpen() {
	auto& c = Core::I();
	auto p = OpenFileDlg();
	if (!p.empty() && c.LoadFile(p.c_str())) {
		std::wstring wp = p;
		size_t ls = wp.find_last_of(L"\\/");
		if (ls != std::wstring::npos) wp = wp.substr(ls + 1);
		g_loadedFileName.clear();
		for (wchar_t wc : wp) g_loadedFileName += (wc < 128) ? (char)wc : '?';

		g_viewAddr = c.EP();
		g_selAddr = g_viewAddr;
		g_hexAddr = c.Base();
		NavClearAll();
		NavPushDisasm(g_viewAddr, "Entry Point");
		NavPushHex(g_hexAddr, "Base");
		g_stringCache.clear();
		g_disPreBufferValid = false;
		g_focusedPanel = FOCUS_DISASM;
		g_displayBase = c.Base();
		g_useCustomBase = false;

		// ÖNEMLİ: Mevcut stringleri tara!
		g_stringStorage.ScanExistingStrings(c.pe, c.mem);

		Log("Loaded: %s Base=%s%llX, EP=%s%llX",
			g_loadedFileName.c_str(), g_showHexPrefix ? "0x" : "", c.Base(),
			g_showHexPrefix ? "0x" : "", c.EP());

		// Bulunan stringleri logla
		if (!g_stringStorage.entries.empty()) {
			Log("Found %zu existing strings in sections", g_stringStorage.entries.size());
			for (auto& e : g_stringStorage.entries) {
				if (e.valid && e.wasExisting) {
					Log("  %s%llX: \"%s\"", g_showHexPrefix ? "0x" : "", e.addr, e.content.c_str());
				}
			}
		}
	}
}

// === SHORTCUTS ===
static void HandleGlobalShortcuts() {
	if (IsAnyDialogOpen()) return;
	ImGuiIO& io = ImGui::GetIO();

	if (io.KeyCtrl && ImGui::IsKeyPressed(ImGuiKey_G, false)) DoGoto();
	if (io.KeyCtrl && ImGui::IsKeyPressed(ImGuiKey_M, false)) {
		if (io.KeyAlt) {
			// CTRL+M+A (veya CTRL+ALT+M) - Free Memory dialog
			g_freeMemDlgOpen = true;
		}
		else {
			DoToggleModuleNames();
		}
	}
	if (io.KeyCtrl && io.KeyShift && ImGui::IsKeyPressed(ImGuiKey_R, false)) DoBuildRefs();
	if (io.KeyCtrl && ImGui::IsKeyPressed(ImGuiKey_Z, false)) DoUndo();
	if (io.KeyCtrl && ImGui::IsKeyPressed(ImGuiKey_Y, false)) DoRedo();
	if (io.KeyAlt && ImGui::IsKeyPressed(ImGuiKey_LeftArrow, false)) DoNavBack();
	if (io.KeyAlt && ImGui::IsKeyPressed(ImGuiKey_RightArrow, false)) DoNavForward();
	if (io.KeyCtrl && ImGui::IsKeyPressed(ImGuiKey_I, false)) g_injDlgOpen = true;

	// CTRL+ALT+A - Allocate Free Memory (alternatif kısayol)
	if (io.KeyCtrl && io.KeyAlt && ImGui::IsKeyPressed(ImGuiKey_A, false)) {
		g_freeMemDlgOpen = true;
	}
}
// === MAIN ===
int WINAPI wWinMain(HINSTANCE hI, HINSTANCE, LPWSTR, int) {
	WNDCLASSEXA wc = { sizeof(wc),CS_CLASSDC,WndProc,0,0,hI,0,LoadCursor(0,IDC_ARROW),0,0,"UAP",0 };
	RegisterClassExA(&wc);
	int sx = GetSystemMetrics(SM_CXSCREEN), sy = GetSystemMetrics(SM_CYSCREEN);
	g_hwnd = CreateWindowExA(0, "UAP", "Ultra Analyzer Pro", WS_POPUP | WS_VISIBLE, (sx - g_w) / 2, (sy - g_h) / 2, g_w, g_h, 0, 0, hI, 0);
	if (!InitD3D(g_hwnd)) return 1;

	IMGUI_CHECKVERSION(); ImGui::CreateContext();
	ImGuiIO& io = ImGui::GetIO(); io.ConfigFlags &= ~ImGuiConfigFlags_NavEnableKeyboard; io.IniFilename = nullptr;
	ImGui_ImplWin32_Init(g_hwnd); ImGui_ImplDX11_Init(g_dev, g_ctx);
	g_fonts.Load(); Setup();

	while (!GUI.WantClose()) {
		MSG m;
		while (PeekMessage(&m, 0, 0, 0, PM_REMOVE)) {
			TranslateMessage(&m);
			DispatchMessage(&m);
			if (m.message == WM_QUIT) GUI.RequestClose();
		}
		if (GUI.WantClose()) break;
		if (g_fonts.NeedRebuild) g_fonts.Rebuild();

		ImGui_ImplDX11_NewFrame();
		ImGui_ImplWin32_NewFrame();
		ImGui::NewFrame();

		GUI.RenderFullFrame();
		HandleGlobalShortcuts();

		// Tüm dialoglar
		RenderGoto();
		RenderSearch();
		RenderEdit();
		RenderCopyDialog();
		RenderRefsPopup();
		RenderDisplaySettings();
		RenderFontDialog();
		RenderColumnSettings();
		RenderHexEditDialog();
		RenderInjectionDlg();
		RenderFreeMemDlg();  // Yeni dialog

		ImGui::Render();
		float cl[4];
		if (g_themeMode == ThemeMode::Dark) { cl[0] = 0.03f; cl[1] = 0.03f; cl[2] = 0.05f; cl[3] = 1; }
		else { cl[0] = 0.94f; cl[1] = 0.94f; cl[2] = 0.94f; cl[3] = 1; }
		g_ctx->OMSetRenderTargets(1, &g_rtv, 0);
		g_ctx->ClearRenderTargetView(g_rtv, cl);
		ImGui_ImplDX11_RenderDrawData(ImGui::GetDrawData());
		g_swap->Present(1, 0);
	}

	GUI.SaveLayout("layout.cfg");
	SaveConfig();
	GUI.Shutdown(); ImGui_ImplDX11_Shutdown(); ImGui_ImplWin32_Shutdown(); ImGui::DestroyContext();
	CleanD3D(); DestroyWindow(g_hwnd); UnregisterClassA("UAP", hI);
	return 0;
}
